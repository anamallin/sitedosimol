function parseListaMinicursos_(minicursosStr) {
  const text = (minicursosStr || '').toString().trim();
  if (!text) return [];

  const titulosMap = {};
  Object.keys(VAGAS_MINICURSOS).forEach(function(titulo) { titulosMap[titulo] = true; });
  Object.keys(CRONOGRAMA_MINICURSOS).forEach(function(titulo) { titulosMap[titulo] = true; });

  const encontrados = Object.keys(titulosMap).map(function(titulo) {
    return {
      titulo: titulo,
      index: text.indexOf(titulo)
    };
  }).filter(function(item) {
    return item.index !== -1;
  }).sort(function(a, b) {
    return a.index - b.index;
  }).map(function(item) {
    return item.titulo;
  });

  if (encontrados.length) return encontrados;

  return text.split(',').map(function(c) { return c.trim(); }).filter(function(c) { return c; });
}

function verificarConflitosMinicursos_(minicursosStr) {
  if (!minicursosStr) return null;
  const listaCursos = parseListaMinicursos_(minicursosStr);
  
  for (var i = 0; i < listaCursos.length; i++) {
    var cursoA = listaCursos[i];
    var schedA = CRONOGRAMA_MINICURSOS[cursoA];
    if (!schedA) continue;
    
    for (var j = i + 1; j < listaCursos.length; j++) {
      var cursoB = listaCursos[j];
      var schedB = CRONOGRAMA_MINICURSOS[cursoB];
      if (!schedB) continue;
      
      if (schedA.period === schedB.period) {
        // Verifica se há pelo menos um dia em comum
        for (var k = 0; k < schedA.days.length; k++) {
          var day = schedA.days[k];
          if (schedB.days.indexOf(day) !== -1) {
            return {
              curso1: cursoA,
              curso2: cursoB,
              dia: day,
              periodo: schedA.period
            };
          }
        }
      }
    }
  }
  return null;
}

function normalizarCpf_(cpf) {
  const digits = (cpf || '').toString().replace(/\D/g, '');
  return digits ? digits.padStart(11, '0') : '';
}

function parseDataPrioridade_(value) {
  if (value instanceof Date && !isNaN(value.getTime())) return value.getTime();
  if (!value) return null;

  const text = value.toString().trim();
  const brMatch = text.match(/^(\d{1,2})\/(\d{1,2})\/(\d{2,4})(?:\s+(\d{1,2}):(\d{2})(?::(\d{2}))?)?/);
  if (brMatch) {
    const year = brMatch[3].length === 2 ? 2000 + Number(brMatch[3]) : Number(brMatch[3]);
    const date = new Date(year, Number(brMatch[2]) - 1, Number(brMatch[1]), Number(brMatch[4] || 0), Number(brMatch[5] || 0), Number(brMatch[6] || 0));
    return isNaN(date.getTime()) ? null : date.getTime();
  }

  const parsed = new Date(text);
  return isNaN(parsed.getTime()) ? null : parsed.getTime();
}

function normalizarTexto_(value) {
  return (value || '').toString()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .trim();
}

function isPagamentoMinicursoPrioritario_(statusPagamento) {
  const text = (statusPagamento || '').toString().trim();
  if (!text) return false;

  const normalized = normalizarTexto_(text);
  if (
    normalized === 'pendente' ||
    normalized === 'nao se aplica' ||
    normalized === 'previa'
  ) {
    return false;
  }

  return (
    normalized.includes('pago') ||
    normalized.includes('confirm') ||
    normalized.includes('aprov') ||
    /^https?:\/\//i.test(text) ||
    normalized.includes('drive.google') ||
    normalized.includes('docs.google')
  );
}

function getStatusPagamentoMinicursoPublico_(statusPagamento) {
  if (isPagamentoMinicursoPrioritario_(statusPagamento)) return 'Pago';
  return statusPagamento || 'Pendente';
}

function getMapaPrioridadeInscricao_(sheetPrincipal) {
  const dataPrincipal = sheetPrincipal.getDataRange().getValues();
  if (dataPrincipal.length < 2) return {};

  const headers = dataPrincipal[0].map(h => h.toString().toLowerCase().trim());
  const cpfCol = headers.indexOf('cpf');
  const dataColHeader = headers.indexOf('data');
  const dataCol = dataColHeader !== -1 ? dataColHeader : 0; // usa a coluna Data; se não existir, cai na primeira coluna
  const mapa = {};

  if (cpfCol === -1) return mapa;

  for (var i = 1; i < dataPrincipal.length; i++) {
    const cpf = normalizarCpf_(dataPrincipal[i][cpfCol]);
    if (!cpf) continue;

    const prioridade = parseDataPrioridade_(dataPrincipal[i][dataCol]);
    if (prioridade !== null && (!mapa[cpf] || prioridade < mapa[cpf])) {
      mapa[cpf] = prioridade;
    }
  }

  return mapa;
}

function calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos) {
  const mapaPrioridade = getMapaPrioridadeInscricao_(sheetPrincipal);
  const dataMinicursos = sheetMinicursos.getDataRange().getValues();
  const cursos = {};
  const porCpf = {};

  for (var i = 1; i < dataMinicursos.length; i++) {
    const row = dataMinicursos[i];
    const cpf = normalizarCpf_(row[2]);
    if (!cpf) continue;
    const nome = row[1] || '';
    const lista = parseListaMinicursos_(row[3]);
    const prioridade = mapaPrioridade[cpf] || null;
    const pagamentoPrioritario = isPagamentoMinicursoPrioritario_(row[7]);
    const ordem = prioridade === null ? 9000000000000 + i : prioridade;

    lista.forEach(function(curso) {
      if (!cursos[curso]) cursos[curso] = [];
      cursos[curso].push({
        curso: curso,
        cpf: cpf,
        nome: nome,
        rowIndex: i,
        prioridade: prioridade,
        pagamentoPrioritario: pagamentoPrioritario,
        ordem: ordem
      });
    });
  }

  Object.keys(cursos).forEach(function(curso) {
    cursos[curso].sort(function(a, b) {
      if (a.pagamentoPrioritario !== b.pagamentoPrioritario) return a.pagamentoPrioritario ? -1 : 1;
      if (a.ordem !== b.ordem) return a.ordem - b.ordem;
      return a.rowIndex - b.rowIndex;
    });

    const limite = getLimiteMinicurso_(curso);
    cursos[curso].forEach(function(item, index) {
      const ranking = {
        curso: curso,
        posicao: index + 1,
        limite: limite,
        inscritos: cursos[curso].length,
        dentroDasVagas: limite === null || index < limite,
        pagamentoPrioritario: item.pagamentoPrioritario,
        prioridadeInscricao: item.prioridade ? new Date(item.prioridade) : null
      };
      if (!porCpf[item.cpf]) porCpf[item.cpf] = [];
      porCpf[item.cpf].push(ranking);
    });
  });

  return { cursos: cursos, porCpf: porCpf };
}

function calcularPreviaRankingMinicursos_(sheetPrincipal, sheetMinicursos, participante) {
  const mapaPrioridade = getMapaPrioridadeInscricao_(sheetPrincipal);
  const dataMinicursos = sheetMinicursos.getDataRange().getValues();
  const cursos = {};
  const porCpf = {};
  const cpfParticipante = normalizarCpf_(participante.cpf);
  let rowParticipanteExistente = null;
  let statusPagamentoParticipante = '';

  for (var i = 1; i < dataMinicursos.length; i++) {
    const row = dataMinicursos[i];
    const cpf = normalizarCpf_(row[2]);
    if (!cpf) continue;
    if (cpf === cpfParticipante) {
      rowParticipanteExistente = i;
      statusPagamentoParticipante = row[7] || '';
      continue;
    }

    const nome = row[1] || '';
    const lista = parseListaMinicursos_(row[3]);
    const prioridade = mapaPrioridade[cpf] || null;
    const pagamentoPrioritario = isPagamentoMinicursoPrioritario_(row[7]);
    const ordem = prioridade === null ? 9000000000000 + i : prioridade;

    lista.forEach(function(curso) {
      if (!cursos[curso]) cursos[curso] = [];
      cursos[curso].push({
        curso: curso,
        cpf: cpf,
        nome: nome,
        rowIndex: i,
        prioridade: prioridade,
        pagamentoPrioritario: pagamentoPrioritario,
        ordem: ordem
      });
    });
  }

  const prioridadeParticipante = mapaPrioridade[cpfParticipante] || null;
  const rowParticipante = rowParticipanteExistente || dataMinicursos.length;
  const pagamentoPrioritarioParticipante = isPagamentoMinicursoPrioritario_(statusPagamentoParticipante);
  const ordemParticipante = prioridadeParticipante === null ? 9000000000000 + rowParticipante : prioridadeParticipante;
  const cursosSelecionados = parseListaMinicursos_(participante.minicursos);

  cursosSelecionados.forEach(function(curso) {
    if (!cursos[curso]) cursos[curso] = [];
    cursos[curso].push({
      curso: curso,
      cpf: cpfParticipante,
      nome: participante.nome || '',
      rowIndex: rowParticipante,
      prioridade: prioridadeParticipante,
      pagamentoPrioritario: pagamentoPrioritarioParticipante,
      ordem: ordemParticipante
    });
  });

  Object.keys(cursos).forEach(function(curso) {
    cursos[curso].sort(function(a, b) {
      if (a.pagamentoPrioritario !== b.pagamentoPrioritario) return a.pagamentoPrioritario ? -1 : 1;
      if (a.ordem !== b.ordem) return a.ordem - b.ordem;
      return a.rowIndex - b.rowIndex;
    });

    const limite = getLimiteMinicurso_(curso);
    cursos[curso].forEach(function(item, index) {
      const ranking = {
        curso: curso,
        posicao: index + 1,
        limite: limite,
        inscritos: cursos[curso].length,
        dentroDasVagas: limite === null || index < limite,
        pagamentoPrioritario: item.pagamentoPrioritario,
        prioridadeInscricao: item.prioridade ? new Date(item.prioridade) : null
      };
      if (!porCpf[item.cpf]) porCpf[item.cpf] = [];
      porCpf[item.cpf].push(ranking);
    });
  });

  return { cursos: cursos, porCpf: porCpf };
}

function isEmailValido_(email) {
  const text = (email || '').toString().trim();
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(text) ? text : '';
}

function escapeHtml_(value) {
  return (value || '').toString()
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

function getMapaContatosParticipantes_(sheetPrincipal, sheetMinicursos) {
  const contatos = {};
  const dataPrincipal = sheetPrincipal.getDataRange().getValues();

  if (dataPrincipal.length > 1) {
    const headersPrincipal = dataPrincipal[0].map(function(h) { return h.toString().toLowerCase().trim(); });
    const cpfCol = headersPrincipal.indexOf('cpf');
    const nomeCol = headersPrincipal.indexOf('nome');
    const emailCol = headersPrincipal.indexOf('email');

    if (cpfCol !== -1) {
      for (var i = 1; i < dataPrincipal.length; i++) {
        const cpf = normalizarCpf_(dataPrincipal[i][cpfCol]);
        if (!cpf) continue;
        contatos[cpf] = {
          nome: nomeCol !== -1 ? dataPrincipal[i][nomeCol] : '',
          email: emailCol !== -1 ? isEmailValido_(dataPrincipal[i][emailCol]) : ''
        };
      }
    }
  }

  const dataMinicursos = sheetMinicursos.getDataRange().getValues();
  for (var j = 1; j < dataMinicursos.length; j++) {
    const cpfMini = normalizarCpf_(dataMinicursos[j][2]);
    if (!cpfMini) continue;
    const emailMini = isEmailValido_(dataMinicursos[j][5]);
    if (!contatos[cpfMini]) contatos[cpfMini] = { nome: '', email: '' };
    contatos[cpfMini].nome = dataMinicursos[j][1] || contatos[cpfMini].nome;
    if (emailMini) contatos[cpfMini].email = emailMini;
  }

  return contatos;
}

function detectarSaidasDasVagas_(rankingAntes, rankingDepois) {
  const saidasPorCpf = {};
  const porCpfDepois = rankingDepois.porCpf || {};

  Object.keys(porCpfDepois).forEach(function(cpf) {
    const rankingsDepois = porCpfDepois[cpf] || [];
    const rankingsAntes = (rankingAntes.porCpf || {})[cpf] || [];

    rankingsDepois.forEach(function(depois) {
      const antes = rankingsAntes.find(function(item) { return item.curso === depois.curso; });
      if (antes && antes.dentroDasVagas && !depois.dentroDasVagas) {
        if (!saidasPorCpf[cpf]) saidasPorCpf[cpf] = [];
        saidasPorCpf[cpf].push(depois);
      }
    });
  });

  return saidasPorCpf;
}

function recalcularStatusPagamentoMinicurso_(statusAtual, tipoInsc, nMinis) {
  if (statusAtual && statusAtual !== 'Pendente' && statusAtual !== 'Não se aplica') {
    return statusAtual;
  }
  return (tipoInsc === 'Inscrito SIMOL' && nMinis <= 2) ? 'Não se aplica' : 'Pendente';
}

function getHeaderIndex_(headers, aliases) {
  for (var i = 0; i < aliases.length; i++) {
    const index = headers.indexOf(aliases[i]);
    if (index !== -1) return index;
  }
  return -1;
}

function isInscricaoPos_(row, tipoCol, programaCol) {
  const tipo = tipoCol !== -1 ? normalizarTexto_(row[tipoCol]) : '';
  const programa = programaCol !== -1 ? normalizarTexto_(row[programaCol]) : '';
  return (
    tipo.includes('pos-graduacao') ||
    tipo.includes('pos graduacao') ||
    tipo.includes('biologia celular e molecular') ||
    !!programa
  );
}

function getOrganizerStats_(sheetPrincipal, sheetMinicursos) {
  const dataPrincipal = sheetPrincipal.getDataRange().getValues();
  const headersPrincipal = dataPrincipal.length ? dataPrincipal[0].map(function(h) { return h.toString().toLowerCase().trim(); }) : [];
  const nomeCol = getHeaderIndex_(headersPrincipal, ['nome']);
  const cpfCol = getHeaderIndex_(headersPrincipal, ['cpf']);
  const tipoCol = getHeaderIndex_(headersPrincipal, ['tipo_inscricao', 'tipo inscrição', 'tipo de inscrição', 'tipo']);
  const programaCol = getHeaderIndex_(headersPrincipal, ['programa_pos', 'programa pós', 'programa pos', 'programa']);
  const comprovanteCol = getHeaderIndex_(headersPrincipal, ['comprovante']);

  const inscricoes = {
    total: 0,
    normal: 0,
    pos: 0,
    pagas: 0,
    pendentes: 0
  };

  for (var i = 1; i < dataPrincipal.length; i++) {
    const row = dataPrincipal[i];
    const temNome = nomeCol !== -1 && row[nomeCol] && row[nomeCol].toString().trim();
    const temCpf = cpfCol !== -1 && normalizarCpf_(row[cpfCol]);
    if (!temNome && !temCpf) continue;

    inscricoes.total++;
    if (isInscricaoPos_(row, tipoCol, programaCol)) inscricoes.pos++;
    else inscricoes.normal++;

    const comprovante = comprovanteCol !== -1 ? (row[comprovanteCol] || '').toString().trim() : '';
    if (comprovante) inscricoes.pagas++;
    else inscricoes.pendentes++;
  }

  const contatos = getMapaContatosParticipantes_(sheetPrincipal, sheetMinicursos);
  const ranking = calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos);
  const cursos = Object.keys(ranking.cursos || {}).sort().map(function(curso) {
    const inscritos = ranking.cursos[curso] || [];
    const limite = getLimiteMinicurso_(curso);
    const dentroDasVagas = limite === null ? inscritos.length : Math.min(inscritos.length, limite);
    const vagasSobrando = limite === null ? null : Math.max(0, limite - inscritos.length);
    const filaEspera = limite === null ? [] : inscritos.slice(limite).map(function(item, index) {
      const contato = contatos[item.cpf] || {};
      return {
        posicao: limite + index + 1,
        nome: item.nome || contato.nome || '',
        cpf: item.cpf || '',
        email: isEmailValido_(contato.email),
        pagamento: item.pagamentoPrioritario ? 'Pago' : 'Pendente'
      };
    });

    return {
      curso: curso,
      limite: limite,
      totalInscritos: inscritos.length,
      dentroDasVagas: dentroDasVagas,
      vagasSobrando: vagasSobrando,
      filaEspera: filaEspera
    };
  });

  return {
    result: 'success',
    generatedAt: new Date(),
    inscricoes: inscricoes,
    minicursos: cursos
  };
}

function normalizarNomeAba_(nome, usados) {
  var base = (nome || 'Minicurso').toString()
    .replace(/[\[\]\*\?\/\\:]/g, '-')
    .replace(/\s+/g, ' ')
    .trim();
  if (!base) base = 'Minicurso';
  base = base.substring(0, 99);

  var nomeFinal = base;
  var contador = 2;
  while (usados[nomeFinal]) {
    var sufixo = ' ' + contador;
    nomeFinal = base.substring(0, 99 - sufixo.length) + sufixo;
    contador++;
  }
  usados[nomeFinal] = true;
  return nomeFinal;
}

function getHeadersComFallback_(data, prefixo) {
  if (!data.length) return [];
  return data[0].map(function(header, index) {
    var text = (header || '').toString().trim();
    return text || (prefixo + '_' + (index + 1));
  });
}

function montarDadosClassificadosMinicursos_(sheetPrincipal, sheetMinicursos) {
  const dataPrincipal = sheetPrincipal.getDataRange().getValues();
  const dataMinicursos = sheetMinicursos.getDataRange().getValues();
  const headersPrincipal = getHeadersComFallback_(dataPrincipal, 'principal');
  const headersMinicursos = getHeadersComFallback_(dataMinicursos, 'minicursos');
  const ranking = calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos);
  const principalPorCpf = {};

  const cpfColPrincipal = headersPrincipal.map(function(h) { return h.toString().toLowerCase().trim(); }).indexOf('cpf');
  if (cpfColPrincipal !== -1) {
    for (var i = 1; i < dataPrincipal.length; i++) {
      const cpfPrincipal = normalizarCpf_(dataPrincipal[i][cpfColPrincipal]);
      if (cpfPrincipal && !principalPorCpf[cpfPrincipal]) {
        principalPorCpf[cpfPrincipal] = dataPrincipal[i];
      }
    }
  }

  const cabecalho = [
    'Minicurso',
    'Posição',
    'Limite de vagas',
    'Total de inscritos no minicurso',
    'Status no ranking',
    'Pagamento prioritário'
  ].concat(
    headersMinicursos.map(function(h) { return 'Minicursos - ' + h; }),
    headersPrincipal.map(function(h) { return 'Inscrição SIMOL - ' + h; })
  );

  const cursos = {};
  Object.keys(ranking.cursos || {}).sort().forEach(function(curso) {
    const inscritos = ranking.cursos[curso] || [];
    const limite = getLimiteMinicurso_(curso);
    const classificados = limite === null ? inscritos : inscritos.slice(0, limite);
    cursos[curso] = {
      limite: limite,
      totalInscritos: inscritos.length,
      linhas: classificados.map(function(item, index) {
        const rowMini = (dataMinicursos[item.rowIndex] || []).slice();
        const rowPrincipal = (principalPorCpf[item.cpf] || []).slice();
        while (rowMini.length < headersMinicursos.length) rowMini.push('');
        while (rowPrincipal.length < headersPrincipal.length) rowPrincipal.push('');
        return [
          curso,
          index + 1,
          limite === null ? 'Sem limite' : limite,
          inscritos.length,
          'Dentro das vagas',
          item.pagamentoPrioritario ? 'Sim' : 'Não'
        ].concat(rowMini, rowPrincipal);
      })
    };
  });

  return {
    cabecalho: cabecalho,
    cursos: cursos
  };
}

function formatarCpfPublico_(cpf) {
  const digits = normalizarCpf_(cpf);
  if (digits.length !== 11) return cpf || '';
  return digits.replace(/(\d{3})(\d{3})(\d{3})(\d{2})/, '$1.$2.$3-$4');
}

function normalizarNomeArquivo_(value) {
  return (value || '').toString()
    .replace(/[\\/:*?"<>|]/g, '-')
    .replace(/\s+/g, ' ')
    .trim()
    .substring(0, 140) || 'arquivo';
}

function getLogoSimolDataUri_() {
  try {
    const response = UrlFetchApp.fetch(LOGO_SIMOL_URL, { muteHttpExceptions: true });
    if (response.getResponseCode() >= 200 && response.getResponseCode() < 300) {
      const blob = response.getBlob();
      return 'data:' + blob.getContentType() + ';base64,' + Utilities.base64Encode(blob.getBytes());
    }
  } catch (error) {
    console.error('Erro ao carregar logo SIMOL: ' + error.toString());
  }
  return '';
}

function gerarPdfsComprovantesMinicursos(options) {
  options = options || {};
  const spreadsheetId = scriptProp.getProperty('key') || SpreadsheetApp.getActiveSpreadsheet().getId();
  const doc = SpreadsheetApp.openById(spreadsheetId);
  const sheetPrincipal = doc.getSheetByName(sheetName);
  const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
  const sheetMinicursos = ssMinicursos.getSheets()[0];
  const ranking = calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos);
  const dataGeracao = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH-mm');
  const baseFolder = DriveApp.getFolderById(FOLDER_ID_MINICURSOS);
  const parentFolder = options.folderId ? DriveApp.getFolderById(options.folderId) : baseFolder.createFolder('Comprovantes minicursos SIMOL - ' + dataGeracao);
  const maxPdfs = Math.max(1, parseInt(options.maxPdfs || '25', 10) || 25);
  const logoDataUri = getLogoSimolDataUri_();
  const resultado = {
    result: 'success',
    folderUrl: parentFolder.getUrl(),
    folderId: parentFolder.getId(),
    totalPdfs: 0,
    puladosExistentes: 0,
    pendentes: 0,
    finalizado: true,
    cursos: []
  };

  const cursosOrdenados = Object.keys(ranking.cursos || {}).sort();
  for (var c = 0; c < cursosOrdenados.length; c++) {
    const curso = cursosOrdenados[c];
    const inscritos = ranking.cursos[curso] || [];
    const limite = getLimiteMinicurso_(curso);
    const classificados = limite === null ? inscritos : inscritos.slice(0, limite);
    const detalhes = getDetalhesMinicurso_(curso);
    const cursoFolder = getOuCriarSubpasta_(parentFolder, normalizarNomeArquivo_(curso).substring(0, 99));
    const arquivosExistentes = getNomesArquivosDaPasta_(cursoFolder);
    const cursoResultado = {
      curso: curso,
      folderUrl: cursoFolder.getUrl(),
      totalPdfs: 0,
      puladosExistentes: 0,
      pendentes: 0
    };

    for (var index = 0; index < classificados.length; index++) {
      const item = classificados[index];
      const rankingItem = {
        posicao: index + 1,
        inscritos: inscritos.length
      };
      const participante = {
        nome: item.nome || 'Participante',
        cpf: item.cpf || ''
      };
      const nomePdf = normalizarNomeArquivo_(
        'Comprovante - ' + curso + ' - ' + participante.nome + ' - ' + normalizarCpf_(participante.cpf)
      ) + '.pdf';

      if (arquivosExistentes[nomePdf]) {
        cursoResultado.puladosExistentes++;
        resultado.puladosExistentes++;
        continue;
      }

      if (resultado.totalPdfs >= maxPdfs) {
        cursoResultado.pendentes++;
        resultado.pendentes++;
        resultado.finalizado = false;
        continue;
      }

      const html = criarHtmlComprovanteMinicurso_(participante, rankingItem, detalhes, logoDataUri);
      const blob = Utilities.newBlob(html, 'text/html', 'comprovante.html')
        .getAs(MimeType.PDF)
        .setName(nomePdf);
      cursoFolder.createFile(blob);
      arquivosExistentes[nomePdf] = true;
      cursoResultado.totalPdfs++;
      resultado.totalPdfs++;
    }

    resultado.cursos.push(cursoResultado);
  }

  if (!resultado.finalizado) {
    resultado.message = 'Ainda ha PDFs pendentes. Execute novamente com o mesmo folderId para continuar.';
  }

  console.log(JSON.stringify(resultado));
  return resultado;

      const data = sheet.getDataRange().getValues();
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      
      let rowToUpdate = -1;
      for (let i = data.length - 1; i > 0; i--) {
        var cpfSheet = (data[i][2] || '').toString().replace(/\D/g, '').padStart(11, '0');
        if (cpfSheet === cpfBusca) { rowToUpdate = i + 1; break; }
      }
      
      if (rowToUpdate !== -1) {
          const rankingAntes = calcularRankingMinicursos_(sheetPrincipal, sheet);
          sheet.getRange(rowToUpdate, 8).setValue(file.getUrl()); 
          SpreadsheetApp.flush();
          const rankingDepois = calcularRankingMinicursos_(sheetPrincipal, sheet);
          enviarEmailsSaidaVagasMinicursos_(sheetPrincipal, sheet, rankingAntes, rankingDepois);
      }
      return ContentService.createTextOutput(JSON.stringify({ result: 'success', fileUrl: file.getUrl() })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 4. UPLOAD DE COMPROVANTE (Inscrição Normal) ======
    else if (action === 'upload') {
      if (isInscricoesSimolEncerradas_()) return inscricoesSimolEncerradasResponse_();

      const sheet = doc.getSheetByName(sheetName);
      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => h.toString().toLowerCase().trim());
      const data = sheet.getDataRange().getValues();
      
      let nomeIndex = headers.indexOf('nome');
      let cpfIndex = headers.indexOf('cpf');
      let compIndex = headers.indexOf('comprovante');
      let emailIndex = headers.indexOf('email');
      
      if (compIndex === -1) { 
          compIndex = headers.length; 
          sheet.getRange(1, compIndex + 1).setValue('comprovante');
      }
      
      let rowToUpdate = -1;
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      if (cpfIndex !== -1 && cpfBusca) {
        for (let i = data.length - 1; i > 0; i--) {
          let cpfSheet = (data[i][cpfIndex] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheet === cpfBusca) {
            rowToUpdate = i + 1;
            break;
          }
        }
      }
      if (rowToUpdate === -1 && nomeIndex !== -1 && e.parameter.nome) {
        for (let i = data.length - 1; i > 0; i--) {
          if (data[i][nomeIndex] && data[i][nomeIndex].toString().trim().toLowerCase() === e.parameter.nome.toString().trim().toLowerCase()) {
            rowToUpdate = i + 1; break;
          }
        }
      }

      let prodIndex = headers.indexOf('produtos');
      if (prodIndex === -1) { 
          prodIndex = headers.length; 
          sheet.getRange(1, prodIndex + 1).setValue('produtos');
          headers.push('produtos'); // Atualiza a lista em memória
      }

      let tamIndex = headers.indexOf('tamanho_camiseta');
      if (tamIndex === -1) { 
          tamIndex = headers.length; 
          sheet.getRange(1, tamIndex + 1).setValue('tamanho_camiseta');
          headers.push('tamanho_camiseta');
      }

      if (rowToUpdate === -1 && !isInscricaoNormalReabertaDisponivel_()) {
        return inscricoesEsgotadasResponse_();
      }

      const folder = DriveApp.getFolderById(FOLDER_ID);
      if (!e.parameter.fileData || !e.parameter.filename) {
        throw new Error('Comprovante não enviado.');
      }
      const fileData = Utilities.base64Decode(e.parameter.fileData);
      if (!fileData || fileData.length === 0) {
        throw new Error('O arquivo do comprovante está vazio.');
      }
      const blob = Utilities.newBlob(fileData, e.parameter.mimeType, e.parameter.nome + " - " + e.parameter.filename);
      const file = folder.createFile(blob);
      let nomeConfirmacao = e.parameter.nome || '';
      let emailConfirmacao = e.parameter.email || e.parameter.EMAIL || '';
      
      if (rowToUpdate !== -1) {
          sheet.getRange(rowToUpdate, compIndex + 1).setValue(file.getUrl());
          if (!nomeConfirmacao && nomeIndex !== -1) {
              nomeConfirmacao = data[rowToUpdate - 1][nomeIndex] || '';
          }
          if (!emailConfirmacao && emailIndex !== -1) {
              emailConfirmacao = data[rowToUpdate - 1][emailIndex] || '';
          }
          if (e.parameter.PRODUTOS) {
              sheet.getRange(rowToUpdate, prodIndex + 1).setValue(e.parameter.PRODUTOS);
          }
          if (e.parameter.TAMANHO_CAMISETA) {
              sheet.getRange(rowToUpdate, tamIndex + 1).setValue(e.parameter.TAMANHO_CAMISETA);
          }
      } else {
          // Cria uma nova linha na planilha pois o inscrito ainda não existe
          const originalHeaders = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
          const nextRow = sheet.getLastRow() + 1;
          const newRowData = originalHeaders.map(function(header) {
            const hOrig = header.toString();
            const hLower = hOrig.toLowerCase().trim();
            if (hLower === 'data') return new Date();
            if (hLower === 'cpf') return "'" + cpfBusca;
            if (hLower === 'comprovante') return file.getUrl();
            if (hLower === 'produtos') return e.parameter.PRODUTOS || '';
            if (hLower === 'tamanho_camiseta') return e.parameter.TAMANHO_CAMISETA || '';
            // Tenta buscar no e.parameter com o nome original do cabeçalho, com o nome minúsculo ou com letras maiúsculas
            return e.parameter[hOrig] || e.parameter[hLower] || e.parameter[hOrig.toUpperCase()] || '';
          });
          sheet.getRange(nextRow, 1, 1, newRowData.length).setValues([newRowData]);
          registrarUsoInscricaoNormalReaberta_();
          if (!emailConfirmacao) {
              const originalEmailIndex = originalHeaders.map(function(header) { return header.toString().toLowerCase().trim(); }).indexOf('email');
              if (originalEmailIndex !== -1) emailConfirmacao = newRowData[originalEmailIndex] || '';
          }
      }
      try {
        enviarEmailConfirmacaoInscricao_(emailConfirmacao, nomeConfirmacao);
      } catch (emailError) {
        console.error('Erro ao enviar confirmação de inscrição para ' + emailConfirmacao + ': ' + emailError.toString());
      }
      return ContentService.createTextOutput(JSON.stringify({ 'result': 'success', 'fileUrl': file.getUrl() })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 5. VERIFICAÇÃO DE INSCRIÇÃO (SITE) ======
    else if (action === 'verificarInscricao') {
      const nomeBusca = (e.parameter.nome || '').trim().toLowerCase();
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const sheetVerif = doc.getSheetByName(sheetName);
      const data = sheetVerif.getDataRange().getValues();
      const headers = data[0].map(h => h.toString().toLowerCase().trim());
      
      const nomeCol = headers.indexOf('nome');
      const cpfCol = headers.indexOf('cpf');
      
      let encontrado = false;

      if (nomeCol !== -1 && cpfCol !== -1) {
        for (var i = 1; i < data.length; i++) {
          var nomeSheet = (data[i][nomeCol] || '').toString().trim().toLowerCase();
          var cpfSheet = (data[i][cpfCol] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (nomeSheet === nomeBusca && cpfSheet === cpfBusca) { encontrado = true; break; }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ encontrado: encontrado })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 5.1. BUSCAR DADOS DO INSCRITO PARA PRODUTOS ======
    else if (action === 'buscarInscritoFesta') {
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const sheetVerif = doc.getSheetByName(sheetName);
      const data = sheetVerif.getDataRange().getValues();
      const headers = data[0].map(function(h) { return h.toString().toLowerCase().trim(); });
      const cpfCol = headers.indexOf('cpf');
      const nomeCol = headers.indexOf('nome');
      const emailCol = headers.indexOf('email');
      const telefoneCol = headers.indexOf('telefone');

      if (!cpfBusca || cpfCol === -1) {
        return ContentService.createTextOutput(JSON.stringify({ encontrado: false })).setMimeType(ContentService.MimeType.JSON);
      }

      for (var i = data.length - 1; i > 0; i--) {
        var cpfSheet = (data[i][cpfCol] || '').toString().replace(/\D/g, '').padStart(11, '0');
        if (cpfSheet === cpfBusca) {
          return ContentService.createTextOutput(JSON.stringify({
            encontrado: true,
            cpf: cpfBusca,
            nome: nomeCol !== -1 ? data[i][nomeCol] || '' : '',
            email: emailCol !== -1 ? data[i][emailCol] || '' : '',
            telefone: telefoneCol !== -1 ? data[i][telefoneCol] || '' : ''
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }

      return ContentService.createTextOutput(JSON.stringify({ encontrado: false })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 5.5. CHECAR SE CPF EXISTE ======
    else if (action === 'checarCpf') {
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const sheetVerif = doc.getSheetByName(sheetName);
      const data = sheetVerif.getDataRange().getValues();
      const headers = data[0].map(h => h.toString().toLowerCase().trim());
      const cpfCol = headers.indexOf('cpf');
      
      let encontrado = false;
      if (cpfCol !== -1) {
        for (var i = 1; i < data.length; i++) {
          var cpfSheet = (data[i][cpfCol] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheet === cpfBusca) { encontrado = true; break; }
        }
      }
      const inscricoesEncerradas = isInscricoesSimolEncerradas_();
      return ContentService.createTextOutput(JSON.stringify({
        encontrado: encontrado,
        inscricoesAbertas: !inscricoesEncerradas && isInscricaoNormalReabertaDisponivel_(),
        vagasRestantes: inscricoesEncerradas ? 0 : getVagasInscricoesNormaisReabertasRestantes_(),
        message: inscricoesEncerradas ? INSCRICOES_SIMOL_ENCERRADAS_MESSAGE : ''
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 5.6. ESTATÍSTICAS DA ÁREA DO ORGANIZADOR ======
    else if (action === 'getOrganizerStats') {
      const codigo = (e.parameter.codigo || '').toString().trim().toUpperCase();
      if (codigo !== ORGANIZADOR_SECRET_CODE) {
        return ContentService.createTextOutput(JSON.stringify({
          result: 'error',
          message: 'Código de organizador inválido.'
        })).setMimeType(ContentService.MimeType.JSON);
      }

      const sheetPrincipal = doc.getSheetByName(sheetName);
      const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
      const sheetMinicursos = ssMinicursos.getSheets()[0];
      return ContentService.createTextOutput(JSON.stringify(
        getOrganizerStats_(sheetPrincipal, sheetMinicursos)
      )).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 5.7. GERAR EXCEL DOS CLASSIFICADOS NOS MINICURSOS ======
    else if (action === 'gerarExcelClassificadosMinicursos') {
      const codigo = (e.parameter.codigo || '').toString().trim().toUpperCase();
      if (codigo !== ORGANIZADOR_SECRET_CODE) {
        return ContentService.createTextOutput(JSON.stringify({
          result: 'error',
          message: 'Código de organizador inválido.'
        })).setMimeType(ContentService.MimeType.JSON);
      }

      return ContentService.createTextOutput(JSON.stringify(
        gerarExcelClassificadosMinicursos()
      )).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 5.8. GERAR PDFs DE COMPROVANTES DOS CLASSIFICADOS NOS MINICURSOS ======
    else if (action === 'gerarPdfsComprovantesMinicursos') {
      const codigo = (e.parameter.codigo || '').toString().trim().toUpperCase();
      if (codigo !== ORGANIZADOR_SECRET_CODE) {
        return ContentService.createTextOutput(JSON.stringify({
          result: 'error',
          message: 'Código de organizador inválido.'
        })).setMimeType(ContentService.MimeType.JSON);
      }

      return ContentService.createTextOutput(JSON.stringify(
        gerarPdfsComprovantesMinicursos({
          folderId: (e.parameter.folderId || '').toString().trim(),
          maxPdfs: e.parameter.maxPdfs || 25
        })
      )).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 6. BUSCAR DADOS COMPLETOS DO PARTICIPANTE (DASHBOARD) ======
    else if (action === 'getParticipantData') {
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const data = {
        inscricao: { status: 'Não encontrada' },
        minicursos: [],
        festa: { inscrito: false },
        submissoes: [],
        concursos: []
      };

      // 6.1 Dados da Inscrição Principal
      const sheetVerif = doc.getSheetByName(sheetName);
      const dataPrincipal = sheetVerif.getDataRange().getValues();
      const headersPrincipal = dataPrincipal[0].map(h => h.toString().toLowerCase().trim());
      const cpfColP = headersPrincipal.indexOf('cpf');
      const compColP = headersPrincipal.indexOf('comprovante');
      
      if (cpfColP !== -1) {
        for (var i = 1; i < dataPrincipal.length; i++) {
          var cpfSheet = (dataPrincipal[i][cpfColP] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheet === cpfBusca) {
            // Verifica se a célula de comprovante está preenchida
            const temComprovante = compColP !== -1 && dataPrincipal[i][compColP] && dataPrincipal[i][compColP].toString().trim() !== '';
            data.inscricao.status = temComprovante ? 'Pago' : 'Pendente';
            break;
          }
        }
      }

      // 6.2 Dados de Minicursos
      try {
        const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
        const sheetMinis = ssMinicursos.getSheets()[0];
        const dataMinis = sheetMinis.getDataRange().getValues();
        const rankingMinis = calcularRankingMinicursos_(sheetVerif, sheetMinis);
        const rankingsCpf = rankingMinis.porCpf[cpfBusca] || [];
        for (var j = 1; j < dataMinis.length; j++) {
          var cpfSheetM = (dataMinis[j][2] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheetM === cpfBusca) {
            const listaCursos = parseListaMinicursos_(dataMinis[j][3]);
            data.minicursos.push({
              lista: dataMinis[j][3], // coluna D: minicursos
              status: getStatusPagamentoMinicursoPublico_(dataMinis[j][7]),  // coluna H: statusPagamento/comprovante
              rankings: listaCursos.map(function(curso) {
                return rankingsCpf.find(function(r) { return r.curso === curso; }) || {
                  curso: curso,
                  posicao: null,
                  limite: getLimiteMinicurso_(curso),
                  inscritos: null,
                  dentroDasVagas: false,
                  prioridadeInscricao: null
                };
              })
            });
          }
        }
      } catch(e) {}

      // 6.3 Dados de produtos
      const sheetFesta = doc.getSheetByName('produtos');
      if (sheetFesta) {
        const dataFesta = sheetFesta.getDataRange().getValues();
        for (var k = 1; k < dataFesta.length; k++) {
          var cpfSheetF = (dataFesta[k][2] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheetF === cpfBusca) {
            data.festa.inscrito = true;
            break;
          }
        }
      }

      // 6.4 Dados de Concursos
      const sheetConcursosDash = doc.getSheetByName('concursos');
      if (sheetConcursosDash) {
        const dataConcursosDash = sheetConcursosDash.getDataRange().getValues();
        if (dataConcursosDash.length > 1) {
          const headersConcursos = dataConcursosDash[0].map(function(h) { return h.toString().toLowerCase().trim(); });
          const dataColC = getHeaderIndex_(headersConcursos, ['data']);
          const concursoColC = getHeaderIndex_(headersConcursos, ['concurso']);
          const cpfColC = getHeaderIndex_(headersConcursos, ['cpf']);
          const vinculoColC = getHeaderIndex_(headersConcursos, ['vinculo', 'vínculo']);
          const origemColC = getHeaderIndex_(headersConcursos, ['origem']);
          const tituloColC = getHeaderIndex_(headersConcursos, ['titulo', 'título']);
          const observacoesColC = getHeaderIndex_(headersConcursos, ['observacoes', 'observações']);
          const videoNomeColC = getHeaderIndex_(headersConcursos, ['video_nome', 'vídeo_nome', 'arquivo_nome']);
          const videoUrlColC = getHeaderIndex_(headersConcursos, ['video_url', 'vídeo_url', 'arquivo_url']);
          const statusColC = getHeaderIndex_(headersConcursos, ['status']);
          const categoriaImagemColC = getHeaderIndex_(headersConcursos, ['categoria_imagem']);
          const legendaImagemColC = getHeaderIndex_(headersConcursos, ['legenda_imagem']);
          const nomeBoloColC = getHeaderIndex_(headersConcursos, ['nome_bolo']);
          const ingredientesBoloColC = getHeaderIndex_(headersConcursos, ['ingredientes_bolo']);
          const alergenicosBoloColC = getHeaderIndex_(headersConcursos, ['alergenicos_bolo']);
          const responsavelBoloColC = getHeaderIndex_(headersConcursos, ['responsavel_bolo']);

          if (cpfColC !== -1) {
            for (var c = dataConcursosDash.length - 1; c > 0; c--) {
              const cpfSheetC = normalizarCpf_(dataConcursosDash[c][cpfColC]);
              if (cpfSheetC === cpfBusca) {
                data.concursos.push({
                  data: dataColC !== -1 ? dataConcursosDash[c][dataColC] : '',
                  concurso: concursoColC !== -1 ? dataConcursosDash[c][concursoColC] : '',
                  vinculo: vinculoColC !== -1 ? dataConcursosDash[c][vinculoColC] : '',
                  origem: origemColC !== -1 ? dataConcursosDash[c][origemColC] : '',
                  titulo: tituloColC !== -1 ? dataConcursosDash[c][tituloColC] : '',
                  observacoes: observacoesColC !== -1 ? dataConcursosDash[c][observacoesColC] : '',
                  video_nome: videoNomeColC !== -1 ? dataConcursosDash[c][videoNomeColC] : '',
                  video_url: videoUrlColC !== -1 ? dataConcursosDash[c][videoUrlColC] : '',
                  categoria_imagem: categoriaImagemColC !== -1 ? dataConcursosDash[c][categoriaImagemColC] : '',
                  legenda_imagem: legendaImagemColC !== -1 ? dataConcursosDash[c][legendaImagemColC] : '',
                  nome_bolo: nomeBoloColC !== -1 ? dataConcursosDash[c][nomeBoloColC] : '',
                  ingredientes_bolo: ingredientesBoloColC !== -1 ? dataConcursosDash[c][ingredientesBoloColC] : '',
                  alergenicos_bolo: alergenicosBoloColC !== -1 ? dataConcursosDash[c][alergenicosBoloColC] : '',
                  responsavel_bolo: responsavelBoloColC !== -1 ? dataConcursosDash[c][responsavelBoloColC] : '',
                  status: statusColC !== -1 && dataConcursosDash[c][statusColC] ? dataConcursosDash[c][statusColC] : 'Inscrito'
                });
              }
            }
          }
        }
      }

      // 6.5 Dados de Submissões
      const sheetSub = doc.getSheetByName('submissoes');
      if (sheetSub) {
        const dataSub = sheetSub.getDataRange().getValues();
        const headersSub = dataSub[0].map(h => h.toString().toLowerCase().trim());
        const dataEnvioColS = headersSub.indexOf('data_envio');
        const cpfColS = headersSub.indexOf('cpf');
        const tituloColS = headersSub.indexOf('titulo');
        const areaTematicaColS = headersSub.indexOf('area_tematica');
        const autoresAfiliacoesColS = headersSub.indexOf('autores_afiliacoes');
        const autorCorrespondenteColS = headersSub.indexOf('autor_correspondente');
        const categoriaAutorColS = headersSub.indexOf('categoria_autor');
        const tipoEstudoColS = headersSub.indexOf('tipo_estudo');
        const tipoEstudoOutroColS = headersSub.indexOf('tipo_estudo_outro');
        const aprovacaoEticaColS = headersSub.indexOf('aprovacao_etica');
        const protocoloEticaColS = headersSub.indexOf('protocolo_etica');
        const arquivoNomeColS = headersSub.indexOf('arquivo_nome');
        const arquivoUrlColS = headersSub.indexOf('arquivo_url');
        const arquivoIdentificadoNomeColS = headersSub.indexOf('arquivo_identificado_nome');
        const arquivoIdentificadoUrlColS = headersSub.indexOf('arquivo_identificado_url');
        const arquivoSemIdentificacaoNomeColS = headersSub.indexOf('arquivo_sem_identificacao_nome');
        const arquivoSemIdentificacaoUrlColS = headersSub.indexOf('arquivo_sem_identificacao_url');
        const origemColS = headersSub.indexOf('origem');
        const revisor1ColS = headersSub.indexOf('aceite_revisor1');
        const revisor2ColS = headersSub.indexOf('aceite_revisor2');
        const statusColS = headersSub.indexOf('status');
        for (var l = 1; l < dataSub.length; l++) {
          var cpfSheetS = (cpfColS !== -1 ? dataSub[l][cpfColS] : '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheetS === cpfBusca) {
            const r1 = revisor1ColS !== -1 ? dataSub[l][revisor1ColS] : '';
            const r2 = revisor2ColS !== -1 ? dataSub[l][revisor2ColS] : '';
            const statusPlanilha = statusColS !== -1 && dataSub[l][statusColS] ? dataSub[l][statusColS] : '';
            let statusSub = statusPlanilha || "Enviado";
            if (r1 === "Sim" && r2 === "Sim") statusSub = "Aprovado";
            else if (r1 === "Não" || r2 === "Não") statusSub = "Não aprovado";
            else if (!statusPlanilha && (r1 || r2)) statusSub = "Em avaliação";

            data.submissoes.push({
              data_envio: dataEnvioColS !== -1 ? dataSub[l][dataEnvioColS] : '',
              titulo: tituloColS !== -1 ? dataSub[l][tituloColS] : '',
              area_tematica: areaTematicaColS !== -1 ? dataSub[l][areaTematicaColS] : '',
              autores_afiliacoes: autoresAfiliacoesColS !== -1 ? dataSub[l][autoresAfiliacoesColS] : '',
              autor_correspondente: autorCorrespondenteColS !== -1 ? dataSub[l][autorCorrespondenteColS] : '',
              categoria_autor: categoriaAutorColS !== -1 ? dataSub[l][categoriaAutorColS] : '',
              tipo_estudo: tipoEstudoColS !== -1 ? dataSub[l][tipoEstudoColS] : '',
              tipo_estudo_outro: tipoEstudoOutroColS !== -1 ? dataSub[l][tipoEstudoOutroColS] : '',
              aprovacao_etica: aprovacaoEticaColS !== -1 ? dataSub[l][aprovacaoEticaColS] : '',
              protocolo_etica: protocoloEticaColS !== -1 ? dataSub[l][protocoloEticaColS] : '',
              arquivo_nome: arquivoNomeColS !== -1 ? dataSub[l][arquivoNomeColS] : '',
              arquivo_url: arquivoUrlColS !== -1 ? dataSub[l][arquivoUrlColS] : '',
              arquivo_identificado_nome: arquivoIdentificadoNomeColS !== -1 ? dataSub[l][arquivoIdentificadoNomeColS] : '',
              arquivo_identificado_url: arquivoIdentificadoUrlColS !== -1 ? dataSub[l][arquivoIdentificadoUrlColS] : '',
              arquivo_sem_identificacao_nome: arquivoSemIdentificacaoNomeColS !== -1 ? dataSub[l][arquivoSemIdentificacaoNomeColS] : '',
              arquivo_sem_identificacao_url: arquivoSemIdentificacaoUrlColS !== -1 ? dataSub[l][arquivoSemIdentificacaoUrlColS] : '',
              origem: origemColS !== -1 ? dataSub[l][origemColS] : '',
              status: statusSub
            });
          }
        }
      }

      return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 7. INSCRIÇÃO NOS CONCURSOS ======
    else if (action === 'inscreverConcurso') {
      const concursosPermitidos = [
        'Minha Tese em 3 Minutos',
        'Concurso de Bolos',
        'Concurso de Imagens de Microscopia'
      ];
      const concurso = (e.parameter.concurso || '').toString().trim();
      const isConcursoImagem = concurso === 'Concurso de Imagens de Microscopia';
      const isConcursoBolo = concurso === 'Concurso de Bolos';
      const nome = (e.parameter.nome || '').toString().trim();
      const cpfLimpo = normalizarCpf_(e.parameter.cpf);
      const email = isEmailValido_(e.parameter.email);
      const telefone = (e.parameter.telefone || '').toString().trim();
      const vinculo = (e.parameter.vinculo || '').toString().trim();
      const origem = (e.parameter.origem || '').toString().trim() || 'Não informado';
      const titulo = (e.parameter.titulo || '').toString().trim();
      const observacoes = (e.parameter.observacoes || '').toString().trim();
      const categoriaImagem = (e.parameter.categoria_imagem || '').toString().trim();
      const legendaImagem = (e.parameter.legenda_imagem || '').toString().trim();
      const termoCessao = (e.parameter.termo_cessao || '').toString().trim();
      const nomeBolo = (e.parameter.nome_bolo || '').toString().trim();
      const ingredientesBolo = (e.parameter.ingredientes_bolo || '').toString().trim();
      const alergenicosBolo = (e.parameter.alergenicos_bolo || '').toString().trim();
      const responsavelBolo = (e.parameter.responsavel_bolo || '').toString().trim();
      const filenameArquivo = (e.parameter.filename || '').toString().trim();
      const fileDataBase64 = (e.parameter.fileData || '').toString().trim();
      const mimeTypeArquivo = e.parameter.mimeType || 'application/octet-stream';

      if (concursosPermitidos.indexOf(concurso) === -1) {
        throw new Error('Concurso inválido.');
      }
      if (!nome || !cpfLimpo || !email || !telefone || !vinculo) {
        throw new Error('Dados obrigatórios da inscrição não enviados.');
      }
      if (!isConcursoBolo && (!filenameArquivo || !fileDataBase64)) {
        throw new Error(isConcursoImagem ? 'Imagem da inscrição não enviada.' : 'Vídeo da inscrição não enviado.');
      }
      if (FOLDER_ID_VIDEOS_CONCURSOS === 'COLE_AQUI_O_ID_DA_PASTA_DE_VIDEOS') {
        throw new Error('Pasta de arquivos dos concursos ainda não configurada no Apps Script.');
      }
      if (isConcursoBolo) {
        const sheetPrincipalBolo = doc.getSheetByName(sheetName);
        const dataPrincipalBolo = sheetPrincipalBolo.getDataRange().getValues();
        const headersPrincipalBolo = dataPrincipalBolo[0].map(function(h) { return h.toString().toLowerCase().trim(); });
        const cpfColBolo = headersPrincipalBolo.indexOf('cpf');
        let participanteVinculado = false;
        if (cpfColBolo !== -1) {
          for (var b = 1; b < dataPrincipalBolo.length; b++) {
            if (normalizarCpf_(dataPrincipalBolo[b][cpfColBolo]) === cpfLimpo) {
              participanteVinculado = true;
              break;
            }
          }
        }
        if (!participanteVinculado) {
          throw new Error('O Concurso de Bolos é exclusivo para participantes vinculados ao evento.');
        }
        if (!titulo) {
          throw new Error('Informe o participante ou grupo responsável pelo bolo.');
        }
        if (!nomeBolo) {
          throw new Error('Informe o nome criativo do bolo.');
        }
        if (!ingredientesBolo) {
          throw new Error('Informe a lista completa de ingredientes do bolo.');
        }
        if (!alergenicosBolo) {
          throw new Error('Informe as restrições alimentares e possíveis alergênicos.');
        }
        if (!responsavelBolo) {
          throw new Error('Informe o responsável pela entrega, identificação e retirada dos materiais.');
        }
      } else if (isConcursoImagem) {
        if (!titulo || contarPalavras_(titulo) > 6) {
          throw new Error('Informe um título da imagem com no máximo 6 palavras.');
        }
        if (!categoriaImagem) {
          throw new Error('Informe a categoria da imagem.');
        }
        if (!legendaImagem || contarPalavras_(legendaImagem) > 100) {
          throw new Error('Informe uma legenda da imagem com no máximo 100 palavras.');
        }
        if (termoCessao !== 'Sim') {
          throw new Error('Confirme o termo de cessão de uso da imagem.');
        }
        if (!/^image\/(jpeg|tiff)$/i.test(mimeTypeArquivo) && !/\.(jpe?g|tiff?)$/i.test(filenameArquivo)) {
          throw new Error('Envie uma imagem em JPEG ou TIFF.');
        }
      } else if (!/^video\//i.test(mimeTypeArquivo)) {
        throw new Error('Envie um arquivo de vídeo válido.');
      }

      let arquivoFile = null;
      let arquivoUrl = '';
      if (!isConcursoBolo) {
        const folderVideosConcursos = DriveApp.getFolderById(FOLDER_ID_VIDEOS_CONCURSOS);
        const fileData = Utilities.base64Decode(fileDataBase64);
        if (!fileData || fileData.length === 0) {
          throw new Error(isConcursoImagem ? 'O arquivo de imagem está vazio.' : 'O arquivo de vídeo está vazio.');
        }
        const arquivoBlob = Utilities.newBlob(fileData, mimeTypeArquivo, 'CONCURSO - ' + concurso + ' - ' + nome + ' - ' + filenameArquivo);
        arquivoFile = folderVideosConcursos.createFile(arquivoBlob);
        arquivoUrl = arquivoFile.getUrl();
      }

      const concursosHeaders = ['Data', 'Concurso', 'Nome', 'CPF', 'Vinculo', 'Origem', 'Email', 'Telefone', 'Titulo', 'Observacoes', 'Video_nome', 'Video_url', 'Status', 'Categoria_imagem', 'Legenda_imagem', 'Termo_cessao', 'Nome_bolo', 'Ingredientes_bolo', 'Alergenicos_bolo', 'Responsavel_bolo'];
      let sheetConcursos = doc.getSheetByName('concursos');
      if (!sheetConcursos) {
        sheetConcursos = doc.insertSheet('concursos');
      }
      if (sheetConcursos.getMaxColumns() < concursosHeaders.length) {
        sheetConcursos.insertColumnsAfter(sheetConcursos.getMaxColumns(), concursosHeaders.length - sheetConcursos.getMaxColumns());
      }
      if (sheetConcursos.getLastRow() === 0) {
        sheetConcursos.appendRow(concursosHeaders);
      } else {
        sheetConcursos.getRange(1, 1, 1, concursosHeaders.length).setValues([concursosHeaders]);
      }

      const dataConcursos = sheetConcursos.getDataRange().getValues();
      let rowToUpdate = -1;
      for (var i = dataConcursos.length - 1; i > 0; i--) {
        const cpfSheet = normalizarCpf_(dataConcursos[i][3]);
        const concursoSheet = (dataConcursos[i][1] || '').toString().trim();
        const categoriaSheet = (dataConcursos[i][13] || '').toString().trim();
        if (cpfSheet === cpfLimpo && concursoSheet === concurso && (!isConcursoImagem || categoriaSheet === categoriaImagem)) {
          rowToUpdate = i + 1;
          break;
        }
      }

      const rowData = [
        new Date(),
        concurso,
        nome,
        "'" + cpfLimpo,
        vinculo,
        origem,
        email,
        telefone,
        titulo,
        observacoes,
        arquivoFile ? arquivoFile.getName() : '',
        arquivoUrl,
        'Inscrito',
        categoriaImagem,
        legendaImagem,
        termoCessao,
        nomeBolo,
        ingredientesBolo,
        alergenicosBolo,
        responsavelBolo
      ];

      if (rowToUpdate !== -1) {
        sheetConcursos.getRange(rowToUpdate, 1, 1, concursosHeaders.length).setValues([rowData]);
      } else {
        sheetConcursos.appendRow(rowData);
      }

      try {
        enviarEmailConfirmacaoConcurso_(email, nome, concurso);
      } catch (emailError) {
        console.error('Erro ao enviar confirmação de inscrição no concurso para ' + email + ': ' + emailError.toString());
      }

      return ContentService.createTextOutput(JSON.stringify({ result: 'success', updated: rowToUpdate !== -1 })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 7.1. BUSCAR DADOS DO INSCRITO PARA CONCURSOS ======
    else if (action === 'buscarInscritoConcurso') {
      const cpfBusca = normalizarCpf_(e.parameter.cpf);
      const sheetVerif = doc.getSheetByName(sheetName);
      const data = sheetVerif.getDataRange().getValues();
      const headers = data[0].map(function(h) { return h.toString().toLowerCase().trim(); });
      const cpfCol = headers.indexOf('cpf');
      const nomeCol = headers.indexOf('nome');
      const emailCol = headers.indexOf('email');
      const telefoneCol = headers.indexOf('telefone');

      if (!cpfBusca || cpfCol === -1) {
        return ContentService.createTextOutput(JSON.stringify({ encontrado: false })).setMimeType(ContentService.MimeType.JSON);
      }

      for (var i = data.length - 1; i > 0; i--) {
        const cpfSheet = normalizarCpf_(data[i][cpfCol]);
        if (cpfSheet === cpfBusca) {
          return ContentService.createTextOutput(JSON.stringify({
            encontrado: true,
            cpf: cpfBusca,
            nome: nomeCol !== -1 ? data[i][nomeCol] || '' : '',
            email: emailCol !== -1 ? data[i][emailCol] || '' : '',
            telefone: telefoneCol !== -1 ? data[i][telefoneCol] || '' : ''
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }

      return ContentService.createTextOutput(JSON.stringify({ encontrado: false })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 8. UPLOAD DE COMPROVANTE (Produtos) ======
    else if (action === 'uploadFesta') {
      const folder = DriveApp.getFolderById(FOLDER_ID);
      if (!e.parameter.fileData || !e.parameter.filename) {
        throw new Error('Comprovante não enviado.');
      }
      const fileData = Utilities.base64Decode(e.parameter.fileData);
      if (!fileData || fileData.length === 0) {
        throw new Error('O arquivo do comprovante está vazio.');
      }
      const tipoProduto = (e.parameter.tipo || '').toString().trim();
      if (!tipoProduto) {
        throw new Error('Tipo do produto não informado.');
      }
      const quantidadeProduto = Math.max(1, parseInt(e.parameter.quantidade || '1', 10) || 1);
      const tipoQuantidade = tipoProduto + ' - quantidade: ' + quantidadeProduto;
      const blob = Utilities.newBlob(fileData, e.parameter.mimeType, "PRODUTO - " + tipoProduto + " - " + e.parameter.nome + " - " + e.parameter.filename);
      const file = folder.createFile(blob);
      let sheetProdutos = doc.getSheetByName('produtos');
      const produtosHeaders = ['Data', 'Nome', 'CPF', 'Tipo_quantidade', 'Valor', 'Comprovante', 'email', 'telefone', 'Observacoes'];
      if (!sheetProdutos) {
        sheetProdutos = doc.insertSheet('produtos');
      }
      if (sheetProdutos.getMaxColumns() < produtosHeaders.length) {
        sheetProdutos.insertColumnsAfter(sheetProdutos.getMaxColumns(), produtosHeaders.length - sheetProdutos.getMaxColumns());
      }
      if (sheetProdutos.getMaxColumns() > produtosHeaders.length) {
        sheetProdutos.deleteColumns(produtosHeaders.length + 1, sheetProdutos.getMaxColumns() - produtosHeaders.length);
      }
      if (sheetProdutos.getLastRow() === 0) {
        sheetProdutos.appendRow(produtosHeaders);
      } else {
        sheetProdutos.getRange(1, 1, 1, produtosHeaders.length).setValues([produtosHeaders]);
      }
      var cpfParamFesta = (e.parameter.cpf || '').toString().replace(/\D/g, '');
      var cpfFesta = cpfParamFesta ? "'" + cpfParamFesta.padStart(11, '0') : '';

      sheetProdutos.appendRow([
        new Date(), 
        e.parameter.nome, 
        cpfFesta, 
        tipoQuantidade, 
        'R$ ' + e.parameter.valor, 
        file.getUrl(),
        e.parameter.email || '',
        e.parameter.telefone || '',
        e.parameter.observacoes || ''
      ]);

      return ContentService.createTextOutput(JSON.stringify({ 'result': 'success', 'fileUrl': file.getUrl() })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 9. INSCRIÇÃO NORMAL (Dados do Formulário) ======
    else {
      return ContentService.createTextOutput(JSON.stringify({
        result: 'error',
        message: 'A inscrição só é gravada após o envio do comprovante.'
      })).setMimeType(ContentService.MimeType.JSON);
    }

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ 'result': 'error', 'error': error.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    if (lockAcquired) lock.releaseLock();
  }
}

/**

function getBrevoApiKeyComprovantes_() {
  var apiKey = PropertiesService
    .getScriptProperties()
    .getProperty('BREVO_API_KEY');

  apiKey = String(apiKey || '').trim();

  if (!apiKey) {
    throw new Error(
      'A propriedade BREVO_API_KEY não foi encontrada nas Propriedades do script.'
    );
  }

  if (apiKey.indexOf('xsmtpsib-') === 0) {
    throw new Error(
      'BREVO_API_KEY contém uma chave SMTP. Use uma chave de API que começa com xkeysib-.'
    );
  }

  return apiKey;
}

/**
 * Envia uma mensagem pela API transacional da Brevo.
 *
 * @param {Object} dados
 * @param {string} dados.email
 * @param {string} dados.nome
 * @param {string} dados.assunto
 * @param {string} dados.html
 * @param {string} dados.texto
 * @param {GoogleAppsScript.Base.Blob} dados.pdfBlob
 * @param {string} dados.nomeArquivo
 * @return {Object}
 */
function enviarEmailBrevoComprovante_(dados) {
  var apiKey = getBrevoApiKeyComprovantes_();

  var payload = {
    sender: {
      name: BREVO_REMETENTE_NOME_COMPROVANTES,
      email: BREVO_REMETENTE_EMAIL_COMPROVANTES
    },
    to: [
      {
        email: dados.email,
        name: dados.nome
      }
    ],
    replyTo: {
      name: BREVO_REPLY_TO_NOME_COMPROVANTES,
      email: BREVO_REPLY_TO_EMAIL_COMPROVANTES
    },
    subject: dados.assunto,
    htmlContent: dados.html,
    textContent: dados.texto,
    attachment: [
      {
        content: Utilities.base64Encode(dados.pdfBlob.getBytes()),
        name: dados.nomeArquivo
      }
    ]
  };

  var resposta = UrlFetchApp.fetch(
    BREVO_API_URL_COMPROVANTES,
    {
      method: 'post',
      contentType: 'application/json',
      headers: {
        'api-key': apiKey,
        'accept': 'application/json'
      },
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    }
  );

  var codigo = resposta.getResponseCode();
  var conteudo = resposta.getContentText();
  var respostaJson = null;

  try {
    respostaJson = JSON.parse(conteudo || '{}');
  } catch (erroJson) {
    respostaJson = null;
  }

  if (codigo < 200 || codigo >= 300) {
    var mensagemBrevo = respostaJson && respostaJson.message
      ? respostaJson.message
      : conteudo;

    var erro = new Error(
      'Brevo HTTP ' + codigo + ': ' + mensagemBrevo
    );

    erro.brevoHttpCode = codigo;
    erro.brevoResponse = conteudo;
    throw erro;
  }

  return {
    httpCode: codigo,
    messageId: respostaJson && respostaJson.messageId
      ? respostaJson.messageId
      : ''
  };
}

function extrairEmailDeTexto_(texto) {

function getMapaContatosSubmissoes_(doc) {
  var contatos = {};
  var sheetPrincipal = doc.getSheetByName(sheetName);

  if (sheetPrincipal) {
    var dataPrincipal = sheetPrincipal.getDataRange().getValues();

    if (dataPrincipal.length > 1) {
      var headersPrincipal = dataPrincipal[0].map(function(h) {
        return h.toString().toLowerCase().trim();
      });
      var cpfCol = headersPrincipal.indexOf('cpf');
      var nomeCol = headersPrincipal.indexOf('nome');
      var emailCol = headersPrincipal.indexOf('email');

      if (cpfCol !== -1) {
        for (var i = 1; i < dataPrincipal.length; i++) {
          var cpf = normalizarCpf_(dataPrincipal[i][cpfCol]);
          var email = emailCol !== -1
            ? isEmailValido_(dataPrincipal[i][emailCol])
            : '';

          if (cpf && email) {
            contatos[cpf] = {
              nome: nomeCol !== -1 ? dataPrincipal[i][nomeCol] : '',
              email: normalizarEmailComprovante_(email)
            };
          }
        }
      }
    }
  }

  try {
    var ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
    var sheetMinicursos = ssMinicursos.getSheets()[0];
    var dataMinicursos = sheetMinicursos.getDataRange().getValues();

    for (var j = 1; j < dataMinicursos.length; j++) {
      var cpfMini = normalizarCpf_(dataMinicursos[j][2]);
      var emailMini = isEmailValido_(dataMinicursos[j][5]);

      if (cpfMini && emailMini) {
        contatos[cpfMini] = {
          nome: dataMinicursos[j][1] || (contatos[cpfMini] && contatos[cpfMini].nome) || '',
          email: normalizarEmailComprovante_(emailMini)
        };
      }
    }
  } catch (erroMinicursos) {
    console.warn(
      'Nao foi possivel consultar contatos da planilha de minicursos: ' +
      erroMinicursos.toString()
    );
  }

  return contatos;
}

/**
 * Envia pela Brevo a confirmacao de recebimento para quem ja submeteu resumo.
 *
 * Como usar:
 * 1. testarEnvioConfirmacoesSubmissoes()
 * 2. enviarConfirmacoesSubmissoesPorEmail({ maxEmails: 300 })
 *
 * @param {Object=} options
 * @return {Object}
 */
function enviarConfirmacoesSubmissoesPorEmail(options) {
  options = options || {};

  var dryRun = options.dryRun === true;
  var maxEmails = parseInt(options.maxEmails || 300, 10);
  var forcarReenvio = options.forcarReenvio === true;

  if (isNaN(maxEmails) || maxEmails <= 0) {
    maxEmails = 300;
  }

  if (!dryRun) {
    getBrevoApiKeyComprovantes_();
  }

  var scriptProp = PropertiesService.getScriptProperties();
  var spreadsheetId = scriptProp.getProperty('key') ||
    SpreadsheetApp.getActiveSpreadsheet().getId();
  var doc = SpreadsheetApp.openById(spreadsheetId);
  var sheetSubmissoes = doc.getSheetByName('submissoes');

  if (!sheetSubmissoes) {
    throw new Error('A aba "submissoes" nao foi encontrada.');
  }

  var data = sheetSubmissoes.getDataRange().getValues();

  if (data.length < 2) {
    return {
      result: 'success',
      dryRun: dryRun,
      enviados: 0,
      message: 'Nao ha submissoes para processar.'
    };
  }

  var headers = data[0].map(function(h) {
    return h.toString().toLowerCase().trim();
  });
  var nomeCol = headers.indexOf('nome');
  var cpfCol = headers.indexOf('cpf');
  var tituloCol = headers.indexOf('titulo');
  var autorCorrespondenteCol = headers.indexOf('autor_correspondente');
  var emailCol = headers.indexOf('email');
  var statusCol = headers.indexOf('status');
  var contatos = getMapaContatosSubmissoes_(doc);
  var enviados = carregarRegistroEnviosSubmissoes_(scriptProp);

  var resultado = {
    result: 'success',
    provedor: 'Brevo',
    dryRun: dryRun,
    linhas_processadas: 0,
    enviados: 0,
    enviados_brevo: 0,
    pulados_ja_enviado: 0,
    pulados_sem_email: 0,
    pulados_sem_dados: 0,
    pendentes: 0,
    erros: 0,
    finalizado: true,
    interrompido_por_limite_brevo: false,
    detalhes: []
  };

  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var linha = i + 1;
    var nome = nomeCol !== -1 ? (row[nomeCol] || '').toString().trim() : '';
    var cpf = cpfCol !== -1 ? normalizarCpf_(row[cpfCol]) : '';
    var titulo = tituloCol !== -1 ? (row[tituloCol] || '').toString().trim() : '';
    var autorCorrespondente = autorCorrespondenteCol !== -1
      ? (row[autorCorrespondenteCol] || '').toString().trim()
      : '';
    var status = statusCol !== -1
      ? (row[statusCol] || '').toString().trim()
      : '';

    resultado.linhas_processadas++;

    if (!nome || !cpf || !titulo) {
      resultado.pulados_sem_dados++;
      resultado.detalhes.push({
        linha: linha,
        status: 'sem_dados_obrigatorios',
        nome: nome,
        cpf: cpf,
        titulo: titulo
      });
      continue;
    }

    var chaveEnvio = cpf + '|' + normalizarTexto_(titulo);

    if (!forcarReenvio && enviados[chaveEnvio]) {
      resultado.pulados_ja_enviado++;
      resultado.detalhes.push({
        linha: linha,
        status: 'ja_enviado',
        nome: nome,
        cpf: cpf,
        titulo: titulo,
        registro: enviados[chaveEnvio]
      });
      continue;
    }

    var email = '';

    if (emailCol !== -1) {
      email = isEmailValido_(row[emailCol]);
    }

    if (!email && contatos[cpf]) {
      email = isEmailValido_(contatos[cpf].email);
    }

    if (!email) {
      email = extrairEmailDeTexto_(autorCorrespondente);
    }

    if (!email) {
      resultado.pulados_sem_email++;
      resultado.detalhes.push({
        linha: linha,
        status: 'sem_email',
        nome: nome,
        cpf: cpf,
        titulo: titulo,
        autor_correspondente: autorCorrespondente
      });
      continue;
    }

    email = normalizarEmailComprovante_(email);

    if (resultado.enviados_brevo >= maxEmails) {
      resultado.pendentes++;
      resultado.finalizado = false;
      resultado.detalhes.push({
        linha: linha,
        status: 'pendente_limite_execucao',
        email: email,
        nome: nome,
        titulo: titulo
      });
      continue;
    }

    try {
      var retornoBrevo = {
        httpCode: null,
        messageId: ''
      };

      if (!dryRun) {
        retornoBrevo = enviarEmailBrevoSemAnexo_({
          email: email,
          nome: nome,
          assunto: 'Submissao recebida - SIMOL 2026',
          html: criarHtmlEmailConfirmacaoSubmissao_(nome, titulo),
          texto: criarTextoEmailConfirmacaoSubmissao_(nome, titulo)
        });
      }

      enviados[chaveEnvio] = dryRun
        ? 'SIMULADO'
        : new Date().toISOString();

      resultado.enviados++;
      resultado.enviados_brevo++;
      resultado.detalhes.push({
        linha: linha,
        status: dryRun ? 'simulado_brevo' : 'enviado_brevo',
        email: email,
        nome: nome,
        cpf: cpf,
        titulo: titulo,
        statusSubmissao: status,
        brevoHttpCode: retornoBrevo.httpCode,
        brevoMessageId: retornoBrevo.messageId
      });

      console.log(
        (dryRun ? '[DRY RUN] ' : '') +
        'Confirmacao de submissao processada: ' +
        email + ' -> ' + titulo +
        (retornoBrevo.messageId
          ? ' | messageId: ' + retornoBrevo.messageId
          : '')
      );

      if (!dryRun) {
        Utilities.sleep(150);
      }

    } catch (erroEmail) {
      resultado.erros++;

      var codigoBrevo = erroEmail.brevoHttpCode || null;
      var textoErro = erroEmail.toString();

      resultado.detalhes.push({
        linha: linha,
        status: 'erro_brevo',
        email: email,
        nome: nome,
        cpf: cpf,
        titulo: titulo,
        brevoHttpCode: codigoBrevo,
        erro: textoErro
      });

      console.error(
        'Erro ao enviar confirmacao de submissao pela Brevo para ' +
        email + ': ' + textoErro
      );

      if (
        codigoBrevo === 429 ||
        /quota|limit|credit|credits|not enough|too many/i.test(textoErro)
      ) {
        resultado.finalizado = false;
        resultado.interrompido_por_limite_brevo = true;
        resultado.pendentes++;
        resultado.message =
          'A Brevo indicou limite de envios, creditos ou requisicoes. ' +
          'Execute novamente quando o limite estiver disponivel.';
        break;
      }
    }
  }

  if (!dryRun) {
    salvarRegistroEnviosSubmissoes_(scriptProp, enviados);
  }

  if (resultado.interrompido_por_limite_brevo) {
    // Mantem a mensagem definida no erro.
  } else if (!resultado.finalizado) {
    resultado.message =
      'O limite de ' + maxEmails +
      ' e-mail(s) por execucao foi atingido. Execute novamente para continuar.';
  } else if (dryRun) {
    resultado.message =
      'Simulacao concluida. Nenhum e-mail foi enviado e nenhuma propriedade foi alterada.';
  } else {
    resultado.message =
      'Envio de confirmacoes de submissao concluido.';
  }

  console.log(JSON.stringify(resultado, null, 2));
  return resultado;
}

/**
 * Carrega o registro de comprovantes enviados.
 *
 * Mantém compatibilidade com a propriedade antiga comprovantes_enviados_v1
 * e também lê o formato dividido em partes, usado para não ultrapassar o
 * limite de tamanho de uma propriedade do Apps Script.
 *
 * @param {GoogleAppsScript.Properties.Properties} scriptProp
 * @return {Object}
 */
function carregarRegistroEnviosComprovantes_(scriptProp) {
  var enviados = {};

  // Formato antigo.
  var jsonAntigo = scriptProp.getProperty('comprovantes_enviados_v1');

  if (jsonAntigo) {
    try {
      var registroAntigo = JSON.parse(jsonAntigo);

      Object.keys(registroAntigo || {}).forEach(function(chave) {
        enviados[chave] = registroAntigo[chave];
      });
    } catch (erroRegistroAntigo) {
      console.warn(
        'Não foi possível interpretar comprovantes_enviados_v1: ' +
        erroRegistroAntigo.toString()
      );
    }
  }

  // Formato atual dividido em partes.
  var quantidadePartes = parseInt(
    scriptProp.getProperty('comprovantes_enviados_v2_partes') || '0',
    10
  );

  if (!isNaN(quantidadePartes) && quantidadePartes > 0) {
    var jsonParticionado = '';

    for (var i = 0; i < quantidadePartes; i++) {
      jsonParticionado +=
        scriptProp.getProperty('comprovantes_enviados_v2_parte_' + i) || '';
    }

    if (jsonParticionado) {
      try {
        var registroParticionado = JSON.parse(jsonParticionado);

        Object.keys(registroParticionado || {}).forEach(function(chave) {
          enviados[chave] = registroParticionado[chave];
        });
      } catch (erroRegistroParticionado) {
        console.warn(
          'Não foi possível interpretar o registro particionado: ' +
          erroRegistroParticionado.toString()
        );
      }
    }
  }

  return enviados;
}


/**
 * Salva o registro de comprovantes enviados em partes menores.
 *
 * @param {GoogleAppsScript.Properties.Properties} scriptProp
 * @param {Object} enviados
 */
function salvarRegistroEnviosComprovantes_(scriptProp, enviados) {
  var json = JSON.stringify(enviados || {});
  var tamanhoParte = 7000;
  var partes = [];

  for (var inicio = 0; inicio < json.length; inicio += tamanhoParte) {
    partes.push(json.substring(inicio, inicio + tamanhoParte));
  }

  if (!partes.length) {
    partes.push('{}');
  }

  var quantidadeAnterior = parseInt(
    scriptProp.getProperty('comprovantes_enviados_v2_partes') || '0',
    10
  );

  if (isNaN(quantidadeAnterior)) {
    quantidadeAnterior = 0;
  }

  for (var i = 0; i < partes.length; i++) {
    scriptProp.setProperty(
      'comprovantes_enviados_v2_parte_' + i,
      partes[i]
    );
  }

  // Remove partes antigas que sobraram após uma eventual redução do registro.
  for (var j = partes.length; j < quantidadeAnterior; j++) {
    scriptProp.deleteProperty(
      'comprovantes_enviados_v2_parte_' + j
    );
  }

  scriptProp.setProperty(
    'comprovantes_enviados_v2_partes',
    String(partes.length)
  );

  // Mantém o formato antigo apenas enquanto ele couber com segurança.
  if (json.length <= 7000) {
    scriptProp.setProperty('comprovantes_enviados_v1', json);
  } else {
    scriptProp.deleteProperty('comprovantes_enviados_v1');
  }
}


/**
 * Monta os mapas de contatos a partir da planilha de minicursos.
 *
 * @return {Object}
 */
function getMapaEmailMinicursos_() {
  var ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
  var sheet = ssMinicursos.getSheets()[0];
  var data = sheet.getDataRange().getValues();

  var mapaCpf = {};
  var mapaNome = {};
  var porMinicurso = {};

  // col 0: Data
  // col 1: Nome
  // col 2: CPF
  // col 3: Minicursos
  // col 4: Telefone
  // col 5: Email
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var cpf = normalizarCpf_(row[2]);
    var nome = (row[1] || '').toString().trim();
    var email = isEmailValido_(row[5]);
    var minicursosStr = (row[3] || '').toString();

    if (!email) {
      continue;
    }

    email = normalizarEmailComprovante_(email);

    if (cpf) {
      mapaCpf[cpf] = {
        nome: nome,
        email: email
      };
    }

    if (nome) {
      var chaveNome = normalizarTexto_(nome);

      if (!mapaNome[chaveNome]) {
        mapaNome[chaveNome] = {
          nome: nome,
          email: email
        };
      }
    }

    var listaMinicursos = parseListaMinicursos_(minicursosStr);

    listaMinicursos.forEach(function(minicurso) {
      var chaveMinicurso = normalizarTexto_(minicurso);

      if (!chaveMinicurso) {
        return;
      }

      if (!porMinicurso[chaveMinicurso]) {
        porMinicurso[chaveMinicurso] = [];
      }

      porMinicurso[chaveMinicurso].push({
        nome: nome,
        nomeNorm: normalizarTexto_(nome),
        email: email
      });
    });
  }

  return {
    porCpf: mapaCpf,
    porNome: mapaNome,
    porMinicurso: porMinicurso
  };
}


/**
 * Procura um contato por nome parcial dentro do mesmo minicurso.
 *
 * @param {string} fragmentoNome
 * @param {string} nomeMinicurso
 * @param {Object} mapaEmail
 * @return {Object|null}
 */
function buscarEmailPorNomeParcial_(fragmentoNome, nomeMinicurso, mapaEmail) {
  var chaveMinicurso = normalizarTexto_(nomeMinicurso);
  var lista = mapaEmail.porMinicurso[chaveMinicurso] || [];

  if (!lista.length || !fragmentoNome) {
    return null;
  }

  var fragmentoNormalizado = normalizarTexto_(fragmentoNome);

  if (!fragmentoNormalizado) {
    return null;
  }

  for (var i = 0; i < lista.length; i++) {
    if (lista[i].nomeNorm === fragmentoNormalizado) {
      return lista[i];
    }
  }

  if (fragmentoNormalizado.length < 6) {
    return null;
  }

  var candidatos = [];

  for (var j = 0; j < lista.length; j++) {
    if (lista[j].nomeNorm.indexOf(fragmentoNormalizado) === 0) {
      candidatos.push(lista[j]);
    }
  }

  if (candidatos.length === 1) {
    return candidatos[0];
  }

  if (candidatos.length > 1) {
    console.warn(
      'Nome parcial ambíguo: "' + fragmentoNome + '" no minicurso "' +
      nomeMinicurso + '". Foram encontrados ' + candidatos.length + ' candidatos.'
    );
  }

  return null;
}


/**
 * Extrai CPF e nome do nome do arquivo PDF.
 *
 * Formato esperado:
 * "Comprovante - <minicurso> - <nome> - <cpf>.pdf"
 *
 * @param {string} nomeArquivo
 * @param {string} nomeMinicurso
 * @return {{cpf: string, nome: string, truncado: boolean}}
 */
function extrairDadosDoNomePdf_(nomeArquivo, nomeMinicurso) {
  var base = String(nomeArquivo || '')
    .replace(/\.pdf$/i, '')
    .trim();

  var prefixoExato = 'Comprovante - ' + nomeMinicurso + ' - ';
  var conteudo = base;

  // Formato normal gerado pelo sistema.
  if (base.indexOf(prefixoExato) === 0) {
    conteudo = base.substring(prefixoExato.length).trim();

  } else if (base.indexOf('Comprovante - ') === 0) {
    // Fallback para títulos de minicurso que sofreram pequenas alterações.
    // Primeiro remove um CPF completo, incompleto ou um hífen vazio no final.
    var resto = base.substring('Comprovante - '.length);
    var restoSemSufixo = resto
      .replace(/\s*-\s*\d{0,11}\s*$/, '')
      .trim();

    // Depois considera o trecho após o último separador como possível nome.
    var ultimoSeparador = restoSemSufixo.lastIndexOf(' - ');

    if (ultimoSeparador !== -1) {
      conteudo = restoSemSufixo
        .substring(ultimoSeparador + 3)
        .trim();
    } else {
      // Quando o nome do arquivo foi cortado ainda no título do minicurso,
      // não há informação suficiente para identificar a pessoa.
      conteudo = '';
    }
  }

  // Também aceita renomeações curtas feitas manualmente:
  // "Nome Completo.pdf", "Nome Completo - CPF.pdf" ou "CPF.pdf".
  var cpf = '';
  var cpfMatch = conteudo.match(/(?:^|\s*-\s*)(\d{11})$/);

  if (cpfMatch) {
    cpf = cpfMatch[1];
  }

  var nomeParte = conteudo
    // Remove CPF completo, CPF incompleto ou apenas o hífen final.
    .replace(/\s*-\s*\d{0,11}\s*$/, '')
    .trim();

  // Caso o nome curto seja composto apenas pelo CPF.
  if (/^\d{11}$/.test(conteudo)) {
    cpf = conteudo;
    nomeParte = '';
  }

  return {
    cpf: cpf,
    nome: nomeParte,
    truncado: !nomeParte && !cpf
  };
}


/**
 * Percorre as subpastas e envia os PDFs pela Brevo.
 *
 * @param {Object=} options
 * @return {Object}
 */
function enviarComprovantesMinicursosPorEmail(options) {
  options = options || {};

  var maxEmailsInformado = parseInt(options.maxEmails, 10);
  var maxEmails = isNaN(maxEmailsInformado) || maxEmailsInformado < 1
    ? 50
    : maxEmailsInformado;

  var forcarReenvio = !!options.forcarReenvio;
  var dryRun = !!options.dryRun;
  var fileId = String(options.fileId || '').trim();
  var scriptProp = PropertiesService.getScriptProperties();

  // Valida a chave antes de percorrer arquivos em uma execução real.
  if (!dryRun) {
    getBrevoApiKeyComprovantes_();
  }

  // ── 1. Localiza a pasta raiz ──
  var pastaRaiz;

  if (options.folderId) {
    pastaRaiz = DriveApp.getFolderById(options.folderId);
  } else {
    var pastaMinicursos = DriveApp.getFolderById(FOLDER_ID_MINICURSOS);
    var subpastas = pastaMinicursos.getFolders();
    var pastaMaisRecente = null;
    var dataMaisRecente = 0;

    while (subpastas.hasNext()) {
      var subpasta = subpastas.next();
      var dataCriacao = subpasta.getDateCreated().getTime();

      if (dataCriacao > dataMaisRecente) {
        dataMaisRecente = dataCriacao;
        pastaMaisRecente = subpasta;
      }
    }

    if (!pastaMaisRecente) {
      console.log('Nenhuma subpasta de comprovantes encontrada.');

      return {
        result: 'error',
        message: 'Nenhuma pasta de comprovantes encontrada.'
      };
    }

    pastaRaiz = pastaMaisRecente;
  }

  // ── 2. Contatos e exclusões ──
  var mapaEmail = getMapaEmailMinicursos_();
  var emailsJaEnviadosManualmente = getMapaEmailsJaEnviadosManualmente_();

  // ── 3. Registro anterior ──
  var enviados = carregarRegistroEnviosComprovantes_(scriptProp);

  // ── 4. Resumo ──
  var resultado = {
    result: 'success',
    provedor: 'Brevo',
    pastaRaiz: pastaRaiz.getName(),
    pastaRaizUrl: pastaRaiz.getUrl(),
    filtroFileId: fileId || null,
    arquivos_selecionados: 0,
    enviados: 0,
    enviados_brevo: 0,
    pulados_ja_enviado: 0,
    pulados_56_ja_enviados: 0,
    pulados_sem_email: 0,
    erros: 0,
    pendentes: 0,
    finalizado: true,
    interrompido_por_limite_brevo: false,
    dryRun: dryRun,
    detalhes: []
  };

  var subpastasMinicurso = pastaRaiz.getFolders();

  cursosLoop:
  while (subpastasMinicurso.hasNext()) {
    var pastaCurso = subpastasMinicurso.next();
    var nomeCurso = pastaCurso.getName();
    var arquivos = pastaCurso.getFilesByType(MimeType.PDF);

    while (arquivos.hasNext()) {
      var arquivo = arquivos.next();

      if (fileId && arquivo.getId() !== fileId) {
        continue;
      }

      resultado.arquivos_selecionados++;

      var nomeArquivo = arquivo.getName();
      var chaveEnvio = pastaCurso.getId() + '/' + nomeArquivo;

      if (!forcarReenvio && enviados[chaveEnvio]) {
        resultado.pulados_ja_enviado++;

        resultado.detalhes.push({
          arquivo: nomeArquivo,
          curso: nomeCurso,
          status: 'ja_enviado',
          registro: enviados[chaveEnvio]
        });

        continue;
      }

      var dadosPdf = extrairDadosDoNomePdf_(nomeArquivo, nomeCurso);
      var cpf = dadosPdf.cpf;
      var nomePdf = dadosPdf.nome;
      var contato = null;
      var encontradoPor = '';

      if (cpf && mapaEmail.porCpf[cpf]) {
        contato = mapaEmail.porCpf[cpf];
        encontradoPor = 'cpf';
      }

      if (!contato && nomePdf) {
        contato = mapaEmail.porNome[normalizarTexto_(nomePdf)] || null;

        if (contato) {
          encontradoPor = 'nome_exato';
        }
      }

      if (!contato && nomePdf) {
        contato = buscarEmailPorNomeParcial_(
          nomePdf,
          nomeCurso,
          mapaEmail
        );

        if (contato) {
          encontradoPor = 'nome_parcial';
        }
      }

      if (!contato || !contato.email) {
        resultado.pulados_sem_email++;

        resultado.detalhes.push({
          arquivo: nomeArquivo,
          curso: nomeCurso,
          status: 'sem_email',
          cpf: cpf,
          nome: nomePdf,
          nomeTruncado: dadosPdf.truncado
        });

        console.warn('E-mail não encontrado para: ' + nomeArquivo);
        continue;
      }

      var emailNormalizado = normalizarEmailComprovante_(contato.email);

      // Proteção explícita para os 56 endereços já enviados pelo SIMOL.
      if (
        !forcarReenvio &&
        emailsJaEnviadosManualmente[emailNormalizado]
      ) {
        resultado.pulados_56_ja_enviados++;

        resultado.detalhes.push({
          arquivo: nomeArquivo,
          curso: nomeCurso,
          status: 'ja_enviado_fora_brevo',
          email: emailNormalizado,
          nome: contato.nome || nomePdf
        });

        // Em execução real, sincroniza o arquivo com o registro persistente.
        if (!dryRun) {
          enviados[chaveEnvio] =
            'JA_ENVIADO_FORA_BREVO_2026-07-10';
        }

        console.log(
          'Ignorado porque já foi enviado fora da Brevo: ' +
          emailNormalizado + ' → ' + nomeArquivo
        );

        continue;
      }

      // O limite vale apenas para novos envios pela Brevo. Arquivos já
      // enviados, endereços da lista dos 56 e casos sem e-mail são tratados
      // antes desta verificação.
      if (resultado.enviados_brevo >= maxEmails) {
        resultado.pendentes++;
        resultado.finalizado = false;

        resultado.detalhes.push({
          arquivo: nomeArquivo,
          curso: nomeCurso,
          status: 'pendente_limite_execucao',
          email: emailNormalizado
        });

        continue;
      }

      var nomeDestinatario = contato.nome || nomePdf || 'participante';

      var htmlBody = criarHtmlEmailComprovante_(
        nomeDestinatario,
        nomeCurso
      );

      var textBody =
        'Olá, ' + nomeDestinatario + '!\n\n' +
        'Sua inscrição no minicurso "' + nomeCurso + '" foi confirmada.\n\n' +
        'Em anexo, você encontra o comprovante oficial em PDF.\n\n' +
        'Data: 20 a 24 de julho de 2026.\n' +
        'Local: SEPT – Setor de Educação Profissional e Tecnológica da UFPR.\n' +
        'Rua Dr. Alcides Vieira Arcoverde, 1225 – Jardim das Américas, Curitiba/PR – CEP 81520-260.\n\n' +
        'Área do Inscrito: ' + AREA_INSCRITO_URL + '\n\n' +
        'Dúvidas: simol.simposiointegrado@gmail.com | @simol.ufpr\n\n' +
        'Atenciosamente,\n' +
        'Comissão Organizadora I SIMOL 2026';

      try {
        var retornoBrevo = {
          httpCode: null,
          messageId: ''
        };

        if (!dryRun) {
          var pdfBlob = arquivo.getBlob().setName(nomeArquivo);

          retornoBrevo = enviarEmailBrevoComprovante_({
            email: emailNormalizado,
            nome: nomeDestinatario,
            assunto:
              '✅ Sua inscrição no minicurso "' +
              nomeCurso +
              '" foi confirmada! | SIMOL 2026',
            html: htmlBody,
            texto: textBody,
            pdfBlob: pdfBlob,
            nomeArquivo: nomeArquivo
          });
        }

        enviados[chaveEnvio] = dryRun
          ? 'SIMULADO'
          : new Date().toISOString();

        resultado.enviados++;
        resultado.enviados_brevo++;

        resultado.detalhes.push({
          arquivo: nomeArquivo,
          curso: nomeCurso,
          status: dryRun ? 'simulado_brevo' : 'enviado_brevo',
          email: emailNormalizado,
          nome: nomeDestinatario,
          encontradoPor: encontradoPor,
          brevoHttpCode: retornoBrevo.httpCode,
          brevoMessageId: retornoBrevo.messageId
        });

        console.log(
          (dryRun ? '[DRY RUN] ' : '') +
          'Processado pela Brevo: ' +
          emailNormalizado + ' → ' + nomeArquivo +
          (retornoBrevo.messageId
            ? ' | messageId: ' + retornoBrevo.messageId
            : '')
        );

        // Pequena pausa para reduzir o risco de limite de requisições.
        if (!dryRun) {
          Utilities.sleep(150);
        }

      } catch (erroEmail) {
        resultado.erros++;

        var codigoBrevo = erroEmail.brevoHttpCode || null;
        var textoErro = erroEmail.toString();

        resultado.detalhes.push({
          arquivo: nomeArquivo,
          curso: nomeCurso,
          status: 'erro_brevo',
          email: emailNormalizado,
          brevoHttpCode: codigoBrevo,
          erro: textoErro
        });

        console.error(
          'Erro ao enviar pela Brevo para ' +
          emailNormalizado + ': ' + textoErro
        );

        // Para quando a Brevo indicar limite, excesso de requisições ou créditos.
        if (
          codigoBrevo === 429 ||
          /quota|limit|credit|credits|not enough|too many/i.test(textoErro)
        ) {
          resultado.finalizado = false;
          resultado.interrompido_por_limite_brevo = true;
          resultado.pendentes++;
          resultado.message =
            'A Brevo indicou limite de envios, créditos ou requisições. ' +
            'Execute novamente quando o limite estiver disponível.';

          break cursosLoop;
        }
      }
    }
  }

  // ── 5. Persiste enviados reais e os 56 sincronizados ──
  if (!dryRun) {
    try {
      salvarRegistroEnviosComprovantes_(scriptProp, enviados);
    } catch (erroSalvarPropriedade) {
      console.warn(
        'Não foi possível salvar o registro de enviados: ' +
        erroSalvarPropriedade.toString()
      );
    }
  }

  if (fileId && resultado.arquivos_selecionados === 0) {
    resultado.result = 'error';
    resultado.finalizado = false;
    resultado.message =
      'O fileId informado não foi encontrado nas subpastas da pasta raiz.';
  } else if (resultado.interrompido_por_limite_brevo) {
    // Mantém a mensagem definida no momento da interrupção.
  } else if (!resultado.finalizado) {
    resultado.message =
      'O limite de ' + maxEmails +
      ' e-mail(s) por execução foi atingido. Execute novamente para continuar.';
  } else if (dryRun) {
    resultado.message =
      'Simulação concluída. Nenhum e-mail foi enviado e nenhuma propriedade foi alterada.';
  } else {
    resultado.message =
      'Execução concluída. Os e-mails anteriores e os 56 endereços informados foram ignorados.';
  }

  console.log(JSON.stringify(resultado, null, 2));
  return resultado;
}

/**
 * Cria/atualiza uma aba com os comprovantes que ainda não podem ser
 * associados com segurança a um participante.
 *
 * COMO USAR:
 * 1. Execute enviarComprovantesMinicursosPorEmail() uma vez com este código.
 *    Os arquivos com nome legível e CPF incompleto serão enviados.
 * 2. Execute gerarRelatorioPendenciasComprovantes().
 * 3. Na aba "Pendências comprovantes", abra cada PDF pelo link e preencha
 *    a coluna "Novo nome completo" com o nome que aparece dentro do PDF.
 * 4. Execute aplicarRenomeacoesPendenciasComprovantes().
 * 5. Execute enviarComprovantesMinicursosPorEmail() novamente.
 *
 * @param {Object=} options
 * @return {Object}
 */
function gerarRelatorioPendenciasComprovantes(options) {
  options = options || {};

  var scriptProp = PropertiesService.getScriptProperties();
  var pastaRaiz = obterPastaRaizPendenciasComprovantes_(options.folderId);
  var mapaEmail = getMapaEmailMinicursos_();
  var emailsJaEnviadosManualmente = getMapaEmailsJaEnviadosManualmente_();
  var enviados = carregarRegistroEnviosComprovantes_(scriptProp);

  var ss = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
  var nomeAba = 'Pendências comprovantes';
  var aba = ss.getSheetByName(nomeAba);

  if (!aba) {
    aba = ss.insertSheet(nomeAba);
  }

  aba.clear();

  var cabecalhos = [[
    'Minicurso',
    'Nome atual do PDF',
    'Link para abrir',
    'File ID',
    'CPF extraído',
    'Nome extraído',
    'Novo nome completo',
    'Status / orientação'
  ]];

  aba.getRange(1, 1, 1, cabecalhos[0].length).setValues(cabecalhos);
  aba.setFrozenRows(1);

  var linhas = [];
  var subpastas = pastaRaiz.getFolders();

  while (subpastas.hasNext()) {
    var pastaCurso = subpastas.next();
    var nomeCurso = pastaCurso.getName();
    var arquivos = pastaCurso.getFilesByType(MimeType.PDF);

    while (arquivos.hasNext()) {
      var arquivo = arquivos.next();
      var nomeArquivo = arquivo.getName();
      var chaveEnvio = pastaCurso.getId() + '/' + nomeArquivo;

      if (enviados[chaveEnvio]) {
        continue;
      }

      var dadosPdf = extrairDadosDoNomePdf_(nomeArquivo, nomeCurso);
      var contato = null;

      if (dadosPdf.cpf && mapaEmail.porCpf[dadosPdf.cpf]) {
        contato = mapaEmail.porCpf[dadosPdf.cpf];
      }

      if (!contato && dadosPdf.nome) {
        contato =
          mapaEmail.porNome[normalizarTexto_(dadosPdf.nome)] ||
          buscarEmailPorNomeParcial_(
            dadosPdf.nome,
            nomeCurso,
            mapaEmail
          );
      }

      // Se já conseguimos identificar, não é uma pendência de nome.
      if (contato && contato.email) {
        var emailNormalizado =
          normalizarEmailComprovante_(contato.email);

        if (emailsJaEnviadosManualmente[emailNormalizado]) {
          continue;
        }

        // Está identificável e será tratado pela função de envio.
        continue;
      }

      linhas.push([
        nomeCurso,
        nomeArquivo,
        arquivo.getUrl(),
        arquivo.getId(),
        dadosPdf.cpf || '',
        dadosPdf.nome || '',
        '',
        'Abra o PDF, confira o participante e preencha "Novo nome completo".'
      ]);
    }
  }

  if (linhas.length) {
    aba.getRange(2, 1, linhas.length, cabecalhos[0].length)
      .setValues(linhas);

    aba.getRange(2, 3, linhas.length, 1)
      .setRichTextValues(
        linhas.map(function(linha) {
          return [
            SpreadsheetApp.newRichTextValue()
              .setText('Abrir PDF')
              .setLinkUrl(linha[2])
              .build()
          ];
        })
      );
  }

  aba.autoResizeColumns(1, cabecalhos[0].length);
  aba.setColumnWidth(1, 260);
  aba.setColumnWidth(2, 420);
  aba.setColumnWidth(3, 110);
  aba.setColumnWidth(7, 260);
  aba.setColumnWidth(8, 360);

  console.log(
    linhas.length +
    ' pendência(s) listada(s) na aba "' +
    nomeAba + '".'
  );

  return {
    result: 'success',
    pastaRaiz: pastaRaiz.getName(),
    aba: nomeAba,
    pendencias: linhas.length,
    planilhaUrl: ss.getUrl()
  };
}


/**
 * Renomeia os PDFs listados na aba "Pendências comprovantes".
 *
 * Preencha a coluna "Novo nome completo" antes de executar.
 * O arquivo passa a se chamar apenas "Nome Completo.pdf"; a pasta já informa
 * o minicurso e o novo extrator aceita esse formato curto.
 *
 * @return {Object}
 */
function aplicarRenomeacoesPendenciasComprovantes() {
  var ss = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
  var nomeAba = 'Pendências comprovantes';
  var aba = ss.getSheetByName(nomeAba);

  if (!aba) {
    throw new Error(
      'A aba "' + nomeAba +
      '" não existe. Execute gerarRelatorioPendenciasComprovantes() primeiro.'
    );
  }

  var ultimaLinha = aba.getLastRow();

  if (ultimaLinha < 2) {
    return {
      result: 'success',
      renomeados: 0,
      message: 'Não há pendências na aba.'
    };
  }

  var dados = aba
    .getRange(2, 1, ultimaLinha - 1, 8)
    .getValues();

  var renomeados = 0;
  var ignorados = 0;
  var erros = 0;

  for (var i = 0; i < dados.length; i++) {
    var fileId = String(dados[i][3] || '').trim();
    var novoNome = String(dados[i][6] || '').trim();
    var statusAtual = String(dados[i][7] || '');

    if (!fileId || !novoNome || /^renomeado/i.test(statusAtual)) {
      ignorados++;
      continue;
    }

    try {
      var nomeSeguro = novoNome
        .replace(/[\\\/:*?"<>|#%{}~]/g, ' ')
        .replace(/\s+/g, ' ')
        .trim();

      if (!nomeSeguro) {
        throw new Error('O novo nome ficou vazio após a limpeza.');
      }

      var arquivo = DriveApp.getFileById(fileId);
      arquivo.setName(nomeSeguro + '.pdf');

      aba.getRange(i + 2, 8).setValue(
        'Renomeado em ' +
        Utilities.formatDate(
          new Date(),
          Session.getScriptTimeZone(),
          'dd/MM/yyyy HH:mm:ss'
        )
      );

      renomeados++;

    } catch (erro) {
      aba.getRange(i + 2, 8).setValue(
        'ERRO: ' + erro.toString()
      );
      erros++;
    }
  }

  console.log(
    JSON.stringify({
      renomeados: renomeados,
      ignorados: ignorados,
      erros: erros
    }, null, 2)
  );

  return {
    result: erros ? 'partial' : 'success',
    renomeados: renomeados,
    ignorados: ignorados,
    erros: erros
  };
}


/**
 * Localiza a pasta raiz usada pelo relatório de pendências.
 *
 * @param {string=} folderId
 * @return {GoogleAppsScript.Drive.Folder}
 */
function obterPastaRaizPendenciasComprovantes_(folderId) {
  if (folderId) {
    return DriveApp.getFolderById(folderId);
  }

  var pastaMinicursos = DriveApp.getFolderById(
    FOLDER_ID_MINICURSOS
  );

  var subpastas = pastaMinicursos.getFolders();
  var pastaMaisRecente = null;
  var dataMaisRecente = 0;

  while (subpastas.hasNext()) {
    var subpasta = subpastas.next();
    var dataCriacao = subpasta.getDateCreated().getTime();

    if (dataCriacao > dataMaisRecente) {
      dataMaisRecente = dataCriacao;
      pastaMaisRecente = subpasta;
    }
  }

  if (!pastaMaisRecente) {
    throw new Error(
      'Nenhuma pasta de comprovantes encontrada em FOLDER_ID_MINICURSOS.'
    );
  }

  return pastaMaisRecente;
}
