function parseListaMinicursos_(minicursosStr) {
  const text = (minicursosStr || '').toString().trim();
  if (!text) return [];

  const titulosMap = {};
  Object.keys(VAGAS_MINICURSOS).forEach(function(titulo) { titulosMap[titulo] = true; });
  Object.keys(CRONOGRAMA_MINICURSOS).forEach(function(titulo) { titulosMap[titulo] = true; });

  const encontrados = Object.keys(titulosMap).map(function(titulo) {
    return {
      titulo: titulo,
      index: text.indexOf(titulo)
    };
  }).filter(function(item) {
    return item.index !== -1;
  }).sort(function(a, b) {
    return a.index - b.index;
  }).map(function(item) {
    return item.titulo;
  });

  if (encontrados.length) return encontrados;

  return text.split(',').map(function(c) { return c.trim(); }).filter(function(c) { return c; });
}

function verificarConflitosMinicursos_(minicursosStr) {
  if (!minicursosStr) return null;
  const listaCursos = parseListaMinicursos_(minicursosStr);
  
  for (var i = 0; i < listaCursos.length; i++) {
    var cursoA = listaCursos[i];
    var schedA = CRONOGRAMA_MINICURSOS[cursoA];
    if (!schedA) continue;
    
    for (var j = i + 1; j < listaCursos.length; j++) {
      var cursoB = listaCursos[j];
      var schedB = CRONOGRAMA_MINICURSOS[cursoB];
      if (!schedB) continue;
      
      if (schedA.period === schedB.period) {
        // Verifica se há pelo menos um dia em comum
        for (var k = 0; k < schedA.days.length; k++) {
          var day = schedA.days[k];
          if (schedB.days.indexOf(day) !== -1) {
            return {
              curso1: cursoA,
              curso2: cursoB,
              dia: day,
              periodo: schedA.period
            };
          }
        }
      }
    }
  }
  return null;
}

function normalizarCpf_(cpf) {
  const digits = (cpf || '').toString().replace(/\D/g, '');
  return digits ? digits.padStart(11, '0') : '';
}

function parseDataPrioridade_(value) {
  if (value instanceof Date && !isNaN(value.getTime())) return value.getTime();
  if (!value) return null;

  const text = value.toString().trim();
  const brMatch = text.match(/^(\d{1,2})\/(\d{1,2})\/(\d{2,4})(?:\s+(\d{1,2}):(\d{2})(?::(\d{2}))?)?/);
  if (brMatch) {
    const year = brMatch[3].length === 2 ? 2000 + Number(brMatch[3]) : Number(brMatch[3]);
    const date = new Date(year, Number(brMatch[2]) - 1, Number(brMatch[1]), Number(brMatch[4] || 0), Number(brMatch[5] || 0), Number(brMatch[6] || 0));
    return isNaN(date.getTime()) ? null : date.getTime();
  }

  const parsed = new Date(text);
  return isNaN(parsed.getTime()) ? null : parsed.getTime();
}

function normalizarTexto_(value) {
  return (value || '').toString()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .trim();
}

function isPagamentoMinicursoPrioritario_(statusPagamento) {
  const text = (statusPagamento || '').toString().trim();
  if (!text) return false;

  const normalized = normalizarTexto_(text);
  if (
    normalized === 'pendente' ||
    normalized === 'nao se aplica' ||
    normalized === 'previa'
  ) {
    return false;
  }

  return (
    normalized.includes('pago') ||
    normalized.includes('confirm') ||
    normalized.includes('aprov') ||
    /^https?:\/\//i.test(text) ||
    normalized.includes('drive.google') ||
    normalized.includes('docs.google')
  );
}

function getStatusPagamentoMinicursoPublico_(statusPagamento) {
  if (isPagamentoMinicursoPrioritario_(statusPagamento)) return 'Pago';
  return statusPagamento || 'Pendente';
}

function getMapaPrioridadeInscricao_(sheetPrincipal) {
  const dataPrincipal = sheetPrincipal.getDataRange().getValues();
  if (dataPrincipal.length < 2) return {};

  const headers = dataPrincipal[0].map(h => h.toString().toLowerCase().trim());
  const cpfCol = headers.indexOf('cpf');
  const dataColHeader = headers.indexOf('data');
  const dataCol = dataColHeader !== -1 ? dataColHeader : 0; // usa a coluna Data; se não existir, cai na primeira coluna
  const mapa = {};

  if (cpfCol === -1) return mapa;

  for (var i = 1; i < dataPrincipal.length; i++) {
    const cpf = normalizarCpf_(dataPrincipal[i][cpfCol]);
    if (!cpf) continue;

    const prioridade = parseDataPrioridade_(dataPrincipal[i][dataCol]);
    if (prioridade !== null && (!mapa[cpf] || prioridade < mapa[cpf])) {
      mapa[cpf] = prioridade;
    }
  }

  return mapa;
}

function calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos) {
  const mapaPrioridade = getMapaPrioridadeInscricao_(sheetPrincipal);
  const dataMinicursos = sheetMinicursos.getDataRange().getValues();
  const cursos = {};
  const porCpf = {};

  for (var i = 1; i < dataMinicursos.length; i++) {
    const row = dataMinicursos[i];
    const cpf = normalizarCpf_(row[2]);
    if (!cpf) continue;
    const nome = row[1] || '';
    const lista = parseListaMinicursos_(row[3]);
    const prioridade = mapaPrioridade[cpf] || null;
    const pagamentoPrioritario = isPagamentoMinicursoPrioritario_(row[7]);
    const ordem = prioridade === null ? 9000000000000 + i : prioridade;

    lista.forEach(function(curso) {
      if (!cursos[curso]) cursos[curso] = [];
      cursos[curso].push({
        curso: curso,
        cpf: cpf,
        nome: nome,
        rowIndex: i,
        prioridade: prioridade,
        pagamentoPrioritario: pagamentoPrioritario,
        ordem: ordem
      });
    });
  }

  Object.keys(cursos).forEach(function(curso) {
    cursos[curso].sort(function(a, b) {
      if (a.pagamentoPrioritario !== b.pagamentoPrioritario) return a.pagamentoPrioritario ? -1 : 1;
      if (a.ordem !== b.ordem) return a.ordem - b.ordem;
      return a.rowIndex - b.rowIndex;
    });

    const limite = getLimiteMinicurso_(curso);
    cursos[curso].forEach(function(item, index) {
      const ranking = {
        curso: curso,
        posicao: index + 1,
        limite: limite,
        inscritos: cursos[curso].length,
        dentroDasVagas: limite === null || index < limite,
        pagamentoPrioritario: item.pagamentoPrioritario,
        prioridadeInscricao: item.prioridade ? new Date(item.prioridade) : null
      };
      if (!porCpf[item.cpf]) porCpf[item.cpf] = [];
      porCpf[item.cpf].push(ranking);
    });
  });

  return { cursos: cursos, porCpf: porCpf };
}

      if (a.pagamentoPrioritario !== b.pagamentoPrioritario) return a.pagamentoPrioritario ? -1 : 1;
      if (a.ordem !== b.ordem) return a.ordem - b.ordem;
      return a.rowIndex - b.rowIndex;
    });

    const limite = getLimiteMinicurso_(curso);
    cursos[curso].forEach(function(item, index) {
      const ranking = {
        curso: curso,
        posicao: index + 1,
        limite: limite,

      porCpf[item.cpf].push(ranking);
    });
  });

  return { cursos: cursos, porCpf: porCpf };

function isEmailValido_(email) {
  const text = (email || '').toString().trim();
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(text) ? text : '';
}

function escapeHtml_(value) {
  return (value || '').toString()
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

function getMapaContatosParticipantes_(sheetPrincipal, sheetMinicursos) {
  const contatos = {};
  const dataPrincipal = sheetPrincipal.getDataRange().getValues();

  if (dataPrincipal.length > 1) {
    const headersPrincipal = dataPrincipal[0].map(function(h) { return h.toString().toLowerCase().trim(); });
    const cpfCol = headersPrincipal.indexOf('cpf');
    const nomeCol = headersPrincipal.indexOf('nome');
    const emailCol = headersPrincipal.indexOf('email');

    if (cpfCol !== -1) {
      for (var i = 1; i < dataPrincipal.length; i++) {
        const cpf = normalizarCpf_(dataPrincipal[i][cpfCol]);
        if (!cpf) continue;
        contatos[cpf] = {
          nome: nomeCol !== -1 ? dataPrincipal[i][nomeCol] : '',
          email: emailCol !== -1 ? isEmailValido_(dataPrincipal[i][emailCol]) : ''
        };
      }
    }
  }

  const dataMinicursos = sheetMinicursos.getDataRange().getValues();
  for (var j = 1; j < dataMinicursos.length; j++) {
    const cpfMini = normalizarCpf_(dataMinicursos[j][2]);
    if (!cpfMini) continue;
    const emailMini = isEmailValido_(dataMinicursos[j][5]);
    if (!contatos[cpfMini]) contatos[cpfMini] = { nome: '', email: '' };
    contatos[cpfMini].nome = dataMinicursos[j][1] || contatos[cpfMini].nome;
    if (emailMini) contatos[cpfMini].email = emailMini;
  }

  return contatos;
}

function detectarSaidasDasVagas_(rankingAntes, rankingDepois) {
  const saidasPorCpf = {};
  const porCpfDepois = rankingDepois.porCpf || {};

  Object.keys(porCpfDepois).forEach(function(cpf) {
    const rankingsDepois = porCpfDepois[cpf] || [];
    const rankingsAntes = (rankingAntes.porCpf || {})[cpf] || [];

    rankingsDepois.forEach(function(depois) {
      const antes = rankingsAntes.find(function(item) { return item.curso === depois.curso; });
      if (antes && antes.dentroDasVagas && !depois.dentroDasVagas) {
        if (!saidasPorCpf[cpf]) saidasPorCpf[cpf] = [];
        saidasPorCpf[cpf].push(depois);
      }
    });
  });

  return saidasPorCpf;
}

function detectarEntradasDasVagas_(rankingAntes, rankingDepois, cursoAlvo) {
  const entradasPorCpf = {};
  const porCpfDepois = rankingDepois.porCpf || {};

  Object.keys(porCpfDepois).forEach(function(cpf) {
    const depois = (porCpfDepois[cpf] || []).find(function(item) { return item.curso === cursoAlvo; });
    if (!depois || !depois.dentroDasVagas) return;

    const antes = ((rankingAntes.porCpf || {})[cpf] || []).find(function(item) { return item.curso === cursoAlvo; });
    if (antes && !antes.dentroDasVagas) {
      entradasPorCpf[cpf] = depois;
    }
  });

  return entradasPorCpf;
}

function calcularRankingFarmacogeneticaComLimite_(sheetPrincipal, sheetMinicursos, limite) {

  } finally {
    VAGAS_MINICURSOS[MINICURSO_FARMACOGENETICA_TITULO] = limiteOriginal;
  }
}

function getHeaderIndex_(headers, aliases) {
  for (var i = 0; i < aliases.length; i++) {
    const index = headers.indexOf(aliases[i]);
    if (index !== -1) return index;
  }
  return -1;
}

function isInscricaoPos_(row, tipoCol, programaCol) {

function getOrganizerStats_(sheetPrincipal, sheetMinicursos) {
  const dataPrincipal = sheetPrincipal.getDataRange().getValues();
  const headersPrincipal = dataPrincipal.length ? dataPrincipal[0].map(function(h) { return h.toString().toLowerCase().trim(); }) : [];
  const nomeCol = getHeaderIndex_(headersPrincipal, ['nome']);
  const cpfCol = getHeaderIndex_(headersPrincipal, ['cpf']);
  const tipoCol = getHeaderIndex_(headersPrincipal, ['tipo_inscricao', 'tipo inscrição', 'tipo de inscrição', 'tipo']);
  const programaCol = getHeaderIndex_(headersPrincipal, ['programa_pos', 'programa pós', 'programa pos', 'programa']);
  const comprovanteCol = getHeaderIndex_(headersPrincipal, ['comprovante']);

  const inscricoes = {
    total: 0,
    normal: 0,
    pos: 0,
    pagas: 0,
    pendentes: 0
  };

  for (var i = 1; i < dataPrincipal.length; i++) {
    const row = dataPrincipal[i];
    const temNome = nomeCol !== -1 && row[nomeCol] && row[nomeCol].toString().trim();
    const temCpf = cpfCol !== -1 && normalizarCpf_(row[cpfCol]);
    if (!temNome && !temCpf) continue;

    inscricoes.total++;
    if (isInscricaoPos_(row, tipoCol, programaCol)) inscricoes.pos++;
    else inscricoes.normal++;

    const comprovante = comprovanteCol !== -1 ? (row[comprovanteCol] || '').toString().trim() : '';
    if (comprovante) inscricoes.pagas++;
    else inscricoes.pendentes++;
  }

  const contatos = getMapaContatosParticipantes_(sheetPrincipal, sheetMinicursos);
  const ranking = calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos);
  const cursos = Object.keys(ranking.cursos || {}).sort().map(function(curso) {
    const inscritos = ranking.cursos[curso] || [];
    const limite = getLimiteMinicurso_(curso);
    const dentroDasVagas = limite === null ? inscritos.length : Math.min(inscritos.length, limite);
    const vagasSobrando = limite === null ? null : Math.max(0, limite - inscritos.length);
    const filaEspera = limite === null ? [] : inscritos.slice(limite).map(function(item, index) {
      const contato = contatos[item.cpf] || {};
      return {
        posicao: limite + index + 1,
        nome: item.nome || contato.nome || '',
        cpf: item.cpf || '',
        email: isEmailValido_(contato.email),
        pagamento: item.pagamentoPrioritario ? 'Pago' : 'Pendente'
      };
    });

    return {
      curso: curso,
      limite: limite,
      totalInscritos: inscritos.length,
      dentroDasVagas: dentroDasVagas,
      vagasSobrando: vagasSobrando,
      filaEspera: filaEspera
    };
  });

  return {
    result: 'success',
    generatedAt: new Date(),
    inscricoes: inscricoes,
    minicursos: cursos
  };
}

function normalizarNomeAba_(nome, usados) {
  var base = (nome || 'Minicurso').toString()
    .replace(/[\[\]\*\?\/\\:]/g, '-')
    .replace(/\s+/g, ' ')
    .trim();
  if (!base) base = 'Minicurso';
  base = base.substring(0, 99);

  var nomeFinal = base;
  var contador = 2;
  while (usados[nomeFinal]) {
    var sufixo = ' ' + contador;
    nomeFinal = base.substring(0, 99 - sufixo.length) + sufixo;
    contador++;
  }
  usados[nomeFinal] = true;
  return nomeFinal;
}

function getHeadersComFallback_(data, prefixo) {
  if (!data.length) return [];
  return data[0].map(function(header, index) {
    var text = (header || '').toString().trim();
    return text || (prefixo + '_' + (index + 1));
  });
}

function montarDadosClassificadosMinicursos_(sheetPrincipal, sheetMinicursos) {
  const dataPrincipal = sheetPrincipal.getDataRange().getValues();
  const dataMinicursos = sheetMinicursos.getDataRange().getValues();
  const headersPrincipal = getHeadersComFallback_(dataPrincipal, 'principal');
  const headersMinicursos = getHeadersComFallback_(dataMinicursos, 'minicursos');
  const ranking = calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos);
  const principalPorCpf = {};

  const cpfColPrincipal = headersPrincipal.map(function(h) { return h.toString().toLowerCase().trim(); }).indexOf('cpf');
  if (cpfColPrincipal !== -1) {
    for (var i = 1; i < dataPrincipal.length; i++) {
      const cpfPrincipal = normalizarCpf_(dataPrincipal[i][cpfColPrincipal]);
      if (cpfPrincipal && !principalPorCpf[cpfPrincipal]) {
        principalPorCpf[cpfPrincipal] = dataPrincipal[i];
      }
    }
  }

  const cabecalho = [
    'Minicurso',
    'Posição',
    'Limite de vagas',
    'Total de inscritos no minicurso',
    'Status no ranking',
    'Pagamento prioritário'
  ].concat(
    headersMinicursos.map(function(h) { return 'Minicursos - ' + h; }),
    headersPrincipal.map(function(h) { return 'Inscrição SIMOL - ' + h; })
  );

  const cursos = {};
  Object.keys(ranking.cursos || {}).sort().forEach(function(curso) {
    const inscritos = ranking.cursos[curso] || [];
    const limite = getLimiteMinicurso_(curso);
    const classificados = limite === null ? inscritos : inscritos.slice(0, limite);
    cursos[curso] = {
      limite: limite,
      totalInscritos: inscritos.length,
      linhas: classificados.map(function(item, index) {
        const rowMini = (dataMinicursos[item.rowIndex] || []).slice();
        const rowPrincipal = (principalPorCpf[item.cpf] || []).slice();
        while (rowMini.length < headersMinicursos.length) rowMini.push('');
        while (rowPrincipal.length < headersPrincipal.length) rowPrincipal.push('');
        return [
          curso,
          index + 1,
          limite === null ? 'Sem limite' : limite,
          inscritos.length,
          'Dentro das vagas',
          item.pagamentoPrioritario ? 'Sim' : 'Não'
        ].concat(rowMini, rowPrincipal);
      })
    };
  });

  return {
    cabecalho: cabecalho,
    cursos: cursos
  };
}

  };

function doPost(e) {
  const lock = LockService.getScriptLock();
  let lockAcquired = false;

  try {
    lock.waitLock(30000);
    lockAcquired = true;
    const action = e.parameter.action;
    const spreadsheetId = scriptProp.getProperty('key') || SpreadsheetApp.getActiveSpreadsheet().getId();
    const doc = SpreadsheetApp.openById(spreadsheetId);

    // ====== 1. INSCRIÇÃO EM MINICURSOS ======
    if (action === 'inscreverMinicursos') {
      const conflito = verificarConflitosMinicursos_(e.parameter.minicursos);
      if (conflito) {
        const diaNome = ['Segunda', 'Terça', 'Quarta', 'Quinta', 'Sexta'][conflito.dia];
        const periodoNome = conflito.periodo === 'morning' ? 'Manhã' : 'Noite';
        return ContentService.createTextOutput(JSON.stringify({
          result: 'error',
          message: 'Conflito de horário detectado: "' + conflito.curso1 + '" e "' + conflito.curso2 + '" ocorrem no mesmo período (' + diaNome + ' - ' + periodoNome + ').'
        })).setMimeType(ContentService.MimeType.JSON);
      }
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const sheetPrincipal = doc.getSheetByName(sheetName);
      const dataPrincipal = sheetPrincipal.getDataRange().getValues();
      const headersPrincipal = dataPrincipal[0].map(h => h.toString().toLowerCase().trim());
      
      const nomeCol = headersPrincipal.indexOf('nome');
      const cpfCol = headersPrincipal.indexOf('cpf');
      const emailCol = headersPrincipal.indexOf('email');
      const telCol = headersPrincipal.indexOf('telefone');

      // Prioriza o que veio no parâmetro (para alunos externos)
      let email = e.parameter.email || "Não cadastrado";
      let tel = e.parameter.telefone || "Não cadastrado";
      let nomeReal = e.parameter.nome;

      // Se for inscrito SIMOL, tenta buscar na planilha principal se não foi enviado
      if (email === "Não cadastrado" || tel === "Não cadastrado") {
        for (var i = 1; i < dataPrincipal.length; i++) {
          var cpfSheet = (dataPrincipal[i][cpfCol] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheet === cpfBusca) {
            nomeReal = dataPrincipal[i][nomeCol];
            if (email === "Não cadastrado") email = emailCol !== -1 ? dataPrincipal[i][emailCol] : "N/A";
            if (tel === "Não cadastrado") tel = telCol !== -1 ? dataPrincipal[i][telCol] : "N/A";
            break;
          }
        }
      }

      const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
      const sheetMinicursos = ssMinicursos.getSheets()[0];
      
      const dataMinicursos = sheetMinicursos.getDataRange().getValues();
      
      // ===== Verifica se já existe inscrição para atualizar durante o período de inscrições =====
      let rowToUpdate = -1;
      for (var k = 1; k < dataMinicursos.length; k++) {
        var cpfExistente = (dataMinicursos[k][2] || '').toString().replace(/\D/g, '').padStart(11, '0');
        if (cpfExistente === cpfBusca) {
          rowToUpdate = k + 1;
          break;
        }
      }

      // Lógica de Pagamento Automático
      const tipoInsc = e.parameter.tipo;
      const nMinis = parseInt(e.parameter.n_minicursos || 0);
      const statusAtual = rowToUpdate !== -1 ? (dataMinicursos[rowToUpdate - 1][7] || 'Pendente') : 'Pendente';
      const statusPagamento = recalcularStatusPagamentoMinicurso_(statusAtual, tipoInsc, nMinis);

      const rankingAntes = calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos);

      if (rowToUpdate !== -1) {
        sheetMinicursos.getRange(rowToUpdate, 1, 1, 8).setValues([[
          tipoInsc,
          nomeReal,
          "'" + cpfBusca,
          e.parameter.minicursos,
          nMinis,
          email,
          tel,
          statusPagamento
        ]]);
      } else {
        sheetMinicursos.appendRow([
          tipoInsc, 
          nomeReal, 
          "'" + cpfBusca,

      enviarEmailsSaidaVagasMinicursos_(sheetPrincipal, sheetMinicursos, rankingAntes, rankingDepois);

    // ====== 1.1. TROCAR MINICURSOS QUANDO ESTIVER NA LISTA DE ESPERA ======
    else if (action === 'atualizarMinicursos') {

        const diaNome = ['Segunda', 'Terça', 'Quarta', 'Quinta', 'Sexta'][conflito.dia];

      }
      const cpfBusca = normalizarCpf_(e.parameter.cpf);
      const sheetPrincipal = doc.getSheetByName(sheetName);
      const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
      const sheetMinicursos = ssMinicursos.getSheets()[0];
      const dataMinicursos = sheetMinicursos.getDataRange().getValues();
      const rankingAtual = calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos);

      let rowToUpdate = -1;
      for (var i = dataMinicursos.length - 1; i > 0; i--) {
        var cpfSheet = normalizarCpf_(dataMinicursos[i][2]);
        if (cpfSheet === cpfBusca) {

          message: 'Inscrição em minicursos não encontrada para este CPF.'

      }

      const nMinis = parseInt(e.parameter.n_minicursos || 0);
      const statusAtual = dataMinicursos[rowToUpdate - 1][7] || 'Pendente';
      const statusPagamento = recalcularStatusPagamentoMinicurso_(statusAtual, tipoInsc, nMinis);

      sheetMinicursos.getRange(rowToUpdate, 4).setValue(e.parameter.minicursos);
      sheetMinicursos.getRange(rowToUpdate, 5).setValue(nMinis);
      sheetMinicursos.getRange(rowToUpdate, 8).setValue(statusPagamento);
      SpreadsheetApp.flush();

      const rankingDepois = calcularRankingMinicursos_(sheetPrincipal, sheetMinicursos);
      enviarEmailsSaidaVagasMinicursos_(sheetPrincipal, sheetMinicursos, rankingAtual, rankingDepois);

      return ContentService.createTextOutput(JSON.stringify({ result: 'success', updated: true })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 1.2. PRÉVIA DE RANKING SEM GRAVAR NA PLANILHA ======
    else if (action === 'previewRankingMinicursos') {
      const conflito = verificarConflitosMinicursos_(e.parameter.minicursos);
      if (conflito) {
        const diaNome = ['Segunda', 'Terça', 'Quarta', 'Quinta', 'Sexta'][conflito.dia];
        const periodoNome = conflito.periodo === 'morning' ? 'Manhã' : 'Noite';
        return ContentService.createTextOutput(JSON.stringify({
          result: 'error',
          message: 'Conflito de horário detectado: "' + conflito.curso1 + '" e "' + conflito.curso2 + '" ocorrem no mesmo período (' + diaNome + ' - ' + periodoNome + ').'
        })).setMimeType(ContentService.MimeType.JSON);
      }
      const cpfBusca = normalizarCpf_(e.parameter.cpf);
      const sheetPrincipal = doc.getSheetByName(sheetName);
      const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
      const sheetMinicursos = ssMinicursos.getSheets()[0];
      const rankingPreview = calcularPreviaRankingMinicursos_(sheetPrincipal, sheetMinicursos, {
        nome: e.parameter.nome,
        cpf: cpfBusca,
        minicursos: e.parameter.minicursos
      });
      const listaCursos = parseListaMinicursos_(e.parameter.minicursos);
      const rankingsCpf = rankingPreview.porCpf[cpfBusca] || [];

      return ContentService.createTextOutput(JSON.stringify({
        result: 'success',
        minicursos: [{
          lista: e.parameter.minicursos || '',
          status: 'Prévia',
          rankings: listaCursos.map(function(curso) {
            return rankingsCpf.find(function(r) { return r.curso === curso; }) || {
              curso: curso,
              posicao: null,
              limite: getLimiteMinicurso_(curso),
              inscritos: null,
              dentroDasVagas: false,
              prioridadeInscricao: null
            };
          })
        }]
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 1.5. CONSULTAR VAGAS DOS MINICURSOS ======
    else if (action === 'getVagasMinicursos') {
      const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
      const sheetMinicursos = ssMinicursos.getSheets()[0];
      const dataMinicursos = sheetMinicursos.getDataRange().getValues();
      
      var contagem = {};
      for (var j = 1; j < dataMinicursos.length; j++) {
        var cursosNaLinha = parseListaMinicursos_(dataMinicursos[j][3]);
        cursosNaLinha.forEach(function(nomeCurso) {
          if (nomeCurso) {
            contagem[nomeCurso] = (contagem[nomeCurso] || 0) + 1;
          }
        });
      }
      
      return ContentService.createTextOutput(JSON.stringify({ 
        result: 'success', 
        vagas: contagem, 
        limites: VAGAS_MINICURSOS
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 2. SUBMISSÃO DE RESUMO ======
    else if (action === 'submeterResumo') {
      const SUBMISSOES_HEADERS = [
        'data_envio',
        'nome',
        'cpf',
        'titulo',
        'area_tematica',
        'autores_afiliacoes',
        'autor_correspondente',
        'categoria_autor',
        'tipo_estudo',
        'tipo_estudo_outro',
        'aprovacao_etica',
        'protocolo_etica',
        'declara_modelo_oficial',
        'declara_regulamento_dados',
        'arquivo_nome',
        'arquivo_url',
        'origem',
        'aceite_revisor1',
        'aceite_revisor2',
        'status',
        'arquivo_identificado_nome',
        'arquivo_identificado_url',
        'arquivo_sem_identificacao_nome',
        'arquivo_sem_identificacao_url'
      ];
      const nomeSubmissao = (e.parameter.nome || '').trim();
      const cpfFormatado = "'" + (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const cpfLimpo = (e.parameter.cpf || '').replace(/\D/g, '');
      const tituloSubmissao = (e.parameter.titulo || '').trim();
      const areaTematica = (e.parameter.area_tematica || '').trim();
      const autoresAfiliacoes = (e.parameter.autores_afiliacoes || '').trim();
      const autorCorrespondente = (e.parameter.autor_correspondente || '').trim();
      const categoriaAutor = (e.parameter.categoria_autor || '').trim();
      const tipoEstudo = (e.parameter.tipo_estudo || '').trim();
      const tipoEstudoOutro = (e.parameter.tipo_estudo_outro || '').trim();
      const aprovacaoEtica = (e.parameter.aprovacao_etica || '').trim();
      const protocoloEtica = (e.parameter.protocolo_etica || '').trim();
      const declaraModeloOficial = e.parameter.declara_modelo_oficial === 'Sim' ? 'Sim' : '';
      const declaraRegulamentoDados = e.parameter.declara_regulamento_dados === 'Sim' ? 'Sim' : '';
      const filenameIdentificado = (e.parameter.filename_identificado || e.parameter.filename || '').trim();
      const fileDataIdentificadoBase64 = (e.parameter.fileData_identificado || e.parameter.fileData || '').trim();
      const mimeTypeIdentificado = e.parameter.mimeType_identificado || e.parameter.mimeType || 'application/octet-stream';
      const filenameSemIdentificacao = (e.parameter.filename_sem_identificacao || '').trim();
      const fileDataSemIdentificacaoBase64 = (e.parameter.fileData_sem_identificacao || '').trim();
      const mimeTypeSemIdentificacao = e.parameter.mimeType_sem_identificacao || 'application/octet-stream';

      if (!nomeSubmissao || !cpfLimpo || !tituloSubmissao || !areaTematica || !autoresAfiliacoes || !autorCorrespondente || !categoriaAutor || !tipoEstudo || !aprovacaoEtica || !declaraModeloOficial || !declaraRegulamentoDados) {
        throw new Error('Dados obrigatórios da submissão não enviados.');
      }
      if (tipoEstudo === 'Outro' && !tipoEstudoOutro) {
        throw new Error('Informe a classificação científica do tipo de estudo.');
      }
      if (aprovacaoEtica !== 'Não aplicável' && !protocoloEtica) {
        throw new Error('Informe o número do protocolo de aprovação ética.');
      }
      if (!fileDataIdentificadoBase64 || !filenameIdentificado) {
        throw new Error('Arquivo identificado da submissão não enviado.');
      }
      if (!fileDataSemIdentificacaoBase64 || !filenameSemIdentificacao) {
        throw new Error('Arquivo sem identificação da submissão não enviado.');
      }
      if (!/\.pdf$/i.test(filenameIdentificado) || !/\.pdf$/i.test(filenameSemIdentificacao)) {
        throw new Error('Os arquivos da submissão devem ser enviados em PDF.');
      }

      const folderSubmissoes = DriveApp.getFolderById(FOLDER_ID_SUBMISSOES);
      const fileDataIdentificado = Utilities.base64Decode(fileDataIdentificadoBase64);
      const fileDataSemIdentificacao = Utilities.base64Decode(fileDataSemIdentificacaoBase64);
      if (!fileDataIdentificado || fileDataIdentificado.length === 0) {
        throw new Error('Arquivo identificado da submissão está vazio.');
      }
      if (!fileDataSemIdentificacao || fileDataSemIdentificacao.length === 0) {
        throw new Error('Arquivo sem identificação da submissão está vazio.');
      }

      const safeTitulo = tituloSubmissao.replace(/[\\/:*?"<>|]/g, '-');
      const blobIdentificado = Utilities.newBlob(
        fileDataIdentificado,
        mimeTypeIdentificado,
        'SUBMISSAO IDENTIFICADA - ' + nomeSubmissao + ' - ' + safeTitulo + ' - ' + filenameIdentificado
      );
      const arquivoIdentificado = folderSubmissoes.createFile(blobIdentificado);
      const arquivoIdentificadoUrl = arquivoIdentificado.getUrl();
      const blobSemIdentificacao = Utilities.newBlob(
        fileDataSemIdentificacao,
        mimeTypeSemIdentificacao,
        'SUBMISSAO SEM IDENTIFICACAO - ' + nomeSubmissao + ' - ' + safeTitulo + ' - ' + filenameSemIdentificacao
      );
      const arquivoSemIdentificacao = folderSubmissoes.createFile(blobSemIdentificacao);
      const arquivoSemIdentificacaoUrl = arquivoSemIdentificacao.getUrl();

      let sheetSubmissoes = doc.getSheetByName('submissoes');
      if (!sheetSubmissoes) {
        sheetSubmissoes = doc.insertSheet('submissoes');
        sheetSubmissoes.appendRow(SUBMISSOES_HEADERS);
      } else {
        const headersAtuais = sheetSubmissoes.getRange(1, 1, 1, Math.max(sheetSubmissoes.getLastColumn(), SUBMISSOES_HEADERS.length)).getValues()[0];
        const precisaAtualizarCabecalho = SUBMISSOES_HEADERS.some((header, index) => headersAtuais[index] !== header);
        if (precisaAtualizarCabecalho) {
          sheetSubmissoes.getRange(1, 1, 1, SUBMISSOES_HEADERS.length).setValues([SUBMISSOES_HEADERS]);
        }
      }
      
      sheetSubmissoes.appendRow([
        new Date(),
        nomeSubmissao,
        cpfFormatado,
        tituloSubmissao,
        areaTematica,
        autoresAfiliacoes,
        autorCorrespondente,
        categoriaAutor,
        tipoEstudo,
        tipoEstudoOutro,
        aprovacaoEtica,
        protocoloEtica,
        declaraModeloOficial,
        declaraRegulamentoDados,
        arquivoIdentificado.getName(),
        arquivoIdentificadoUrl,
        e.parameter.origem || 'Inscrito SIMOL',
        '', // aceite_revisor1
        '', // aceite_revisor2
        'Enviado',
        arquivoIdentificado.getName(),
        arquivoIdentificadoUrl,
        arquivoSemIdentificacao.getName(),
        arquivoSemIdentificacaoUrl
      ]);

      return ContentService.createTextOutput(JSON.stringify({
        result: 'success',
        fileUrlIdentificado: arquivoIdentificadoUrl,
        fileUrlSemIdentificacao: arquivoSemIdentificacaoUrl
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 3. UPLOAD DE COMPROVANTE (MINICURSOS) ======
    else if (action === 'uploadMinicurso') {
      const folder = DriveApp.getFolderById(FOLDER_ID_MINICURSOS);
      const fileData = Utilities.base64Decode(e.parameter.fileData);
      const blob = Utilities.newBlob(fileData, e.parameter.mimeType, "MINICURSO - " + e.parameter.nome + " - " + e.parameter.filename);
      const file = folder.createFile(blob);
      
      const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
      const sheet = ssMinicursos.getSheets()[0];
      const sheetPrincipal = doc.getSheetByName(sheetName);
      const data = sheet.getDataRange().getValues();
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      
      let rowToUpdate = -1;
      for (let i = data.length - 1; i > 0; i--) {
        var cpfSheet = (data[i][2] || '').toString().replace(/\D/g, '').padStart(11, '0');
        if (cpfSheet === cpfBusca) { rowToUpdate = i + 1; break; }
      }
      
      if (rowToUpdate !== -1) {
          const rankingAntes = calcularRankingMinicursos_(sheetPrincipal, sheet);
          sheet.getRange(rowToUpdate, 8).setValue(file.getUrl()); 
          SpreadsheetApp.flush();
          const rankingDepois = calcularRankingMinicursos_(sheetPrincipal, sheet);
          enviarEmailsSaidaVagasMinicursos_(sheetPrincipal, sheet, rankingAntes, rankingDepois);
      }
      return ContentService.createTextOutput(JSON.stringify({ result: 'success', fileUrl: file.getUrl() })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 4. UPLOAD DE COMPROVANTE (Inscrição Normal) ======
    else if (action === 'upload') {
      if (isInscricoesSimolEncerradas_()) return inscricoesSimolEncerradasResponse_();

      const sheet = doc.getSheetByName(sheetName);
      const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(h => h.toString().toLowerCase().trim());
      const data = sheet.getDataRange().getValues();
      
      let nomeIndex = headers.indexOf('nome');
      let cpfIndex = headers.indexOf('cpf');
      let compIndex = headers.indexOf('comprovante');
      let emailIndex = headers.indexOf('email');
      
      if (compIndex === -1) { 
          compIndex = headers.length; 
          sheet.getRange(1, compIndex + 1).setValue('comprovante');
      }
      
      let rowToUpdate = -1;
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      if (cpfIndex !== -1 && cpfBusca) {
        for (let i = data.length - 1; i > 0; i--) {
          let cpfSheet = (data[i][cpfIndex] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheet === cpfBusca) {
            rowToUpdate = i + 1;
            break;
          }
        }
      }
      if (rowToUpdate === -1 && nomeIndex !== -1 && e.parameter.nome) {
        for (let i = data.length - 1; i > 0; i--) {
          if (data[i][nomeIndex] && data[i][nomeIndex].toString().trim().toLowerCase() === e.parameter.nome.toString().trim().toLowerCase()) {
            rowToUpdate = i + 1; break;
          }
        }
      }

      let prodIndex = headers.indexOf('produtos');
      if (prodIndex === -1) { 
          prodIndex = headers.length; 
          sheet.getRange(1, prodIndex + 1).setValue('produtos');
          headers.push('produtos'); // Atualiza a lista em memória
      }

      let tamIndex = headers.indexOf('tamanho_camiseta');
      if (tamIndex === -1) { 
          tamIndex = headers.length; 
          sheet.getRange(1, tamIndex + 1).setValue('tamanho_camiseta');
          headers.push('tamanho_camiseta');
      }

      if (rowToUpdate === -1 && !isInscricaoNormalReabertaDisponivel_()) {
        return inscricoesEsgotadasResponse_();
      }

      const folder = DriveApp.getFolderById(FOLDER_ID);
      if (!e.parameter.fileData || !e.parameter.filename) {
        throw new Error('Comprovante não enviado.');
      }
      const fileData = Utilities.base64Decode(e.parameter.fileData);
      if (!fileData || fileData.length === 0) {
        throw new Error('O arquivo do comprovante está vazio.');
      }
      const blob = Utilities.newBlob(fileData, e.parameter.mimeType, e.parameter.nome + " - " + e.parameter.filename);
      const file = folder.createFile(blob);
      let nomeConfirmacao = e.parameter.nome || '';
      let emailConfirmacao = e.parameter.email || e.parameter.EMAIL || '';
      
      if (rowToUpdate !== -1) {
          sheet.getRange(rowToUpdate, compIndex + 1).setValue(file.getUrl());
          if (!nomeConfirmacao && nomeIndex !== -1) {
              nomeConfirmacao = data[rowToUpdate - 1][nomeIndex] || '';
          }
          if (!emailConfirmacao && emailIndex !== -1) {
              emailConfirmacao = data[rowToUpdate - 1][emailIndex] || '';
          }
          if (e.parameter.PRODUTOS) {
              sheet.getRange(rowToUpdate, prodIndex + 1).setValue(e.parameter.PRODUTOS);
          }
          if (e.parameter.TAMANHO_CAMISETA) {
              sheet.getRange(rowToUpdate, tamIndex + 1).setValue(e.parameter.TAMANHO_CAMISETA);
          }
      } else {
          // Cria uma nova linha na planilha pois o inscrito ainda não existe
          const originalHeaders = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
          const nextRow = sheet.getLastRow() + 1;
          const newRowData = originalHeaders.map(function(header) {
            const hOrig = header.toString();
            const hLower = hOrig.toLowerCase().trim();
            if (hLower === 'data') return new Date();
            if (hLower === 'cpf') return "'" + cpfBusca;
            if (hLower === 'comprovante') return file.getUrl();
            if (hLower === 'produtos') return e.parameter.PRODUTOS || '';
            if (hLower === 'tamanho_camiseta') return e.parameter.TAMANHO_CAMISETA || '';
            // Tenta buscar no e.parameter com o nome original do cabeçalho, com o nome minúsculo ou com letras maiúsculas
            return e.parameter[hOrig] || e.parameter[hLower] || e.parameter[hOrig.toUpperCase()] || '';
          });
          sheet.getRange(nextRow, 1, 1, newRowData.length).setValues([newRowData]);
          registrarUsoInscricaoNormalReaberta_();
          if (!emailConfirmacao) {
              const originalEmailIndex = originalHeaders.map(function(header) { return header.toString().toLowerCase().trim(); }).indexOf('email');
              if (originalEmailIndex !== -1) emailConfirmacao = newRowData[originalEmailIndex] || '';
          }
      }
      try {
        enviarEmailConfirmacaoInscricao_(emailConfirmacao, nomeConfirmacao);
      } catch (emailError) {
        console.error('Erro ao enviar confirmação de inscrição para ' + emailConfirmacao + ': ' + emailError.toString());
      }
      return ContentService.createTextOutput(JSON.stringify({ 'result': 'success', 'fileUrl': file.getUrl() })).setMimeType(ContentService.MimeType.JSON);

    // ====== 5. VERIFICAÇÃO DE INSCRIÇÃO (SITE) ======
    else if (action === 'verificarInscricao') {

      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const sheetVerif = doc.getSheetByName(sheetName);
      const data = sheetVerif.getDataRange().getValues();
      const headers = data[0].map(h => h.toString().toLowerCase().trim());
      
      const nomeCol = headers.indexOf('nome');
      const cpfCol = headers.indexOf('cpf');
      
      let encontrado = false;

      if (nomeCol !== -1 && cpfCol !== -1) {
        for (var i = 1; i < data.length; i++) {
          var nomeSheet = (data[i][nomeCol] || '').toString().trim().toLowerCase();
          var cpfSheet = (data[i][cpfCol] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (nomeSheet === nomeBusca && cpfSheet === cpfBusca) { encontrado = true; break; }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ encontrado: encontrado })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 5.1. BUSCAR DADOS DO INSCRITO PARA PRODUTOS ======
    else if (action === 'buscarInscritoFesta') {
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const sheetVerif = doc.getSheetByName(sheetName);
      const data = sheetVerif.getDataRange().getValues();
      const headers = data[0].map(function(h) { return h.toString().toLowerCase().trim(); });
      const cpfCol = headers.indexOf('cpf');
      const nomeCol = headers.indexOf('nome');
      const emailCol = headers.indexOf('email');
      const telefoneCol = headers.indexOf('telefone');

      if (!cpfBusca || cpfCol === -1) {
        return ContentService.createTextOutput(JSON.stringify({ encontrado: false })).setMimeType(ContentService.MimeType.JSON);
      }

      for (var i = data.length - 1; i > 0; i--) {
        var cpfSheet = (data[i][cpfCol] || '').toString().replace(/\D/g, '').padStart(11, '0');
        if (cpfSheet === cpfBusca) {
          return ContentService.createTextOutput(JSON.stringify({
            encontrado: true,
            cpf: cpfBusca,
            nome: nomeCol !== -1 ? data[i][nomeCol] || '' : '',
            email: emailCol !== -1 ? data[i][emailCol] || '' : '',
            telefone: telefoneCol !== -1 ? data[i][telefoneCol] || '' : ''
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }

      return ContentService.createTextOutput(JSON.stringify({ encontrado: false })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 5.5. CHECAR SE CPF EXISTE ======
    else if (action === 'checarCpf') {
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const sheetVerif = doc.getSheetByName(sheetName);
      const data = sheetVerif.getDataRange().getValues();
      const headers = data[0].map(h => h.toString().toLowerCase().trim());
      const cpfCol = headers.indexOf('cpf');
      
      let encontrado = false;
      if (cpfCol !== -1) {
        for (var i = 1; i < data.length; i++) {
          var cpfSheet = (data[i][cpfCol] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheet === cpfBusca) { encontrado = true; break; }
        }
      }
      const inscricoesEncerradas = isInscricoesSimolEncerradas_();
      return ContentService.createTextOutput(JSON.stringify({
        encontrado: encontrado,
        inscricoesAbertas: !inscricoesEncerradas && isInscricaoNormalReabertaDisponivel_(),
        vagasRestantes: inscricoesEncerradas ? 0 : getVagasInscricoesNormaisReabertasRestantes_(),
        message: inscricoesEncerradas ? INSCRICOES_SIMOL_ENCERRADAS_MESSAGE : ''
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 5.6. ESTATÍSTICAS DA ÁREA DO ORGANIZADOR ======
    else if (action === 'getOrganizerStats') {
      const codigo = (e.parameter.codigo || '').toString().trim().toUpperCase();
      if (codigo !== ORGANIZADOR_SECRET_CODE) {
        return ContentService.createTextOutput(JSON.stringify({
          result: 'error',

        })).setMimeType(ContentService.MimeType.JSON);

      const sheetPrincipal = doc.getSheetByName(sheetName);

    else if (action === 'notificarEntradaFarmacogenetica') {

          message: 'CÃ³digo de organizador invÃ¡lido.'

      }

      const sheetPrincipal = doc.getSheetByName(sheetName);
      const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);

      const resultado = enviarEmailsEntradaVagasFarmacogenetica_(sheetPrincipal, sheetMinicursos);

      return ContentService.createTextOutput(JSON.stringify(resultado)).setMimeType(ContentService.MimeType.JSON);

    // ====== 5.8. GERAR EXCEL DOS CLASSIFICADOS NOS MINICURSOS ======
    else if (action === 'gerarExcelClassificadosMinicursos') {
      const codigo = (e.parameter.codigo || '').toString().trim().toUpperCase();
      if (codigo !== ORGANIZADOR_SECRET_CODE) {
        return ContentService.createTextOutput(JSON.stringify({
          result: 'error',
          message: 'Código de organizador inválido.'
        })).setMimeType(ContentService.MimeType.JSON);

    }

    // ====== 6. BUSCAR DADOS COMPLETOS DO PARTICIPANTE (DASHBOARD) ======
    else if (action === 'getParticipantData') {
      const cpfBusca = (e.parameter.cpf || '').replace(/\D/g, '').padStart(11, '0');
      const data = {
        inscricao: { status: 'Não encontrada' },
        minicursos: [],
        festa: { inscrito: false },
        submissoes: [],
        concursos: []
      };

      // 6.1 Dados da Inscrição Principal
      const sheetVerif = doc.getSheetByName(sheetName);
      const dataPrincipal = sheetVerif.getDataRange().getValues();
      const headersPrincipal = dataPrincipal[0].map(h => h.toString().toLowerCase().trim());
      const cpfColP = headersPrincipal.indexOf('cpf');
      const compColP = headersPrincipal.indexOf('comprovante');
      
      if (cpfColP !== -1) {
        for (var i = 1; i < dataPrincipal.length; i++) {
          var cpfSheet = (dataPrincipal[i][cpfColP] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheet === cpfBusca) {
            // Verifica se a célula de comprovante está preenchida
            const temComprovante = compColP !== -1 && dataPrincipal[i][compColP] && dataPrincipal[i][compColP].toString().trim() !== '';
            data.inscricao.status = temComprovante ? 'Pago' : 'Pendente';
            break;
          }
        }
      }

      // 6.2 Dados de Minicursos
      try {
        const ssMinicursos = SpreadsheetApp.openById(ID_PLANILHA_MINICURSOS);
        const sheetMinis = ssMinicursos.getSheets()[0];
        const dataMinis = sheetMinis.getDataRange().getValues();
        const rankingMinis = calcularRankingMinicursos_(sheetVerif, sheetMinis);
        const rankingsCpf = rankingMinis.porCpf[cpfBusca] || [];
        for (var j = 1; j < dataMinis.length; j++) {
          var cpfSheetM = (dataMinis[j][2] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheetM === cpfBusca) {
            const listaCursos = parseListaMinicursos_(dataMinis[j][3]);
            data.minicursos.push({
              lista: dataMinis[j][3], // coluna D: minicursos
              status: getStatusPagamentoMinicursoPublico_(dataMinis[j][7]),  // coluna H: statusPagamento/comprovante
              rankings: listaCursos.map(function(curso) {
                return rankingsCpf.find(function(r) { return r.curso === curso; }) || {
                  curso: curso,
                  posicao: null,
                  limite: getLimiteMinicurso_(curso),
                  inscritos: null,
                  dentroDasVagas: false,
                  prioridadeInscricao: null
                };
              })
            });
          }
        }
      } catch(e) {}

      // 6.3 Dados de produtos
      const sheetFesta = doc.getSheetByName('produtos');
      if (sheetFesta) {
        const dataFesta = sheetFesta.getDataRange().getValues();
        for (var k = 1; k < dataFesta.length; k++) {
          var cpfSheetF = (dataFesta[k][2] || '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheetF === cpfBusca) {
            data.festa.inscrito = true;
            break;
          }
        }
      }

      // 6.4 Dados de Concursos
      const sheetConcursosDash = doc.getSheetByName('concursos');
      if (sheetConcursosDash) {
        const dataConcursosDash = sheetConcursosDash.getDataRange().getValues();
        if (dataConcursosDash.length > 1) {
          const headersConcursos = dataConcursosDash[0].map(function(h) { return h.toString().toLowerCase().trim(); });
          const dataColC = getHeaderIndex_(headersConcursos, ['data']);
          const concursoColC = getHeaderIndex_(headersConcursos, ['concurso']);
          const cpfColC = getHeaderIndex_(headersConcursos, ['cpf']);
          const vinculoColC = getHeaderIndex_(headersConcursos, ['vinculo', 'vínculo']);
          const origemColC = getHeaderIndex_(headersConcursos, ['origem']);
          const tituloColC = getHeaderIndex_(headersConcursos, ['titulo', 'título']);
          const observacoesColC = getHeaderIndex_(headersConcursos, ['observacoes', 'observações']);
          const videoNomeColC = getHeaderIndex_(headersConcursos, ['video_nome', 'vídeo_nome', 'arquivo_nome']);
          const videoUrlColC = getHeaderIndex_(headersConcursos, ['video_url', 'vídeo_url', 'arquivo_url']);
          const statusColC = getHeaderIndex_(headersConcursos, ['status']);
          const categoriaImagemColC = getHeaderIndex_(headersConcursos, ['categoria_imagem']);
          const legendaImagemColC = getHeaderIndex_(headersConcursos, ['legenda_imagem']);
          const nomeBoloColC = getHeaderIndex_(headersConcursos, ['nome_bolo']);
          const ingredientesBoloColC = getHeaderIndex_(headersConcursos, ['ingredientes_bolo']);
          const alergenicosBoloColC = getHeaderIndex_(headersConcursos, ['alergenicos_bolo']);
          const responsavelBoloColC = getHeaderIndex_(headersConcursos, ['responsavel_bolo']);

          if (cpfColC !== -1) {
            for (var c = dataConcursosDash.length - 1; c > 0; c--) {
              const cpfSheetC = normalizarCpf_(dataConcursosDash[c][cpfColC]);
              if (cpfSheetC === cpfBusca) {
                data.concursos.push({
                  data: dataColC !== -1 ? dataConcursosDash[c][dataColC] : '',
                  concurso: concursoColC !== -1 ? dataConcursosDash[c][concursoColC] : '',
                  vinculo: vinculoColC !== -1 ? dataConcursosDash[c][vinculoColC] : '',
                  origem: origemColC !== -1 ? dataConcursosDash[c][origemColC] : '',
                  titulo: tituloColC !== -1 ? dataConcursosDash[c][tituloColC] : '',
                  observacoes: observacoesColC !== -1 ? dataConcursosDash[c][observacoesColC] : '',
                  video_nome: videoNomeColC !== -1 ? dataConcursosDash[c][videoNomeColC] : '',
                  video_url: videoUrlColC !== -1 ? dataConcursosDash[c][videoUrlColC] : '',
                  categoria_imagem: categoriaImagemColC !== -1 ? dataConcursosDash[c][categoriaImagemColC] : '',
                  legenda_imagem: legendaImagemColC !== -1 ? dataConcursosDash[c][legendaImagemColC] : '',
                  nome_bolo: nomeBoloColC !== -1 ? dataConcursosDash[c][nomeBoloColC] : '',
                  ingredientes_bolo: ingredientesBoloColC !== -1 ? dataConcursosDash[c][ingredientesBoloColC] : '',
                  alergenicos_bolo: alergenicosBoloColC !== -1 ? dataConcursosDash[c][alergenicosBoloColC] : '',
                  responsavel_bolo: responsavelBoloColC !== -1 ? dataConcursosDash[c][responsavelBoloColC] : '',
                  status: statusColC !== -1 && dataConcursosDash[c][statusColC] ? dataConcursosDash[c][statusColC] : 'Inscrito'
                });
              }
            }
          }
        }
      }

      // 6.5 Dados de Submissões
      const sheetSub = doc.getSheetByName('submissoes');
      if (sheetSub) {
        const dataSub = sheetSub.getDataRange().getValues();
        const headersSub = dataSub[0].map(h => h.toString().toLowerCase().trim());
        const dataEnvioColS = headersSub.indexOf('data_envio');
        const cpfColS = headersSub.indexOf('cpf');
        const tituloColS = headersSub.indexOf('titulo');
        const areaTematicaColS = headersSub.indexOf('area_tematica');
        const autoresAfiliacoesColS = headersSub.indexOf('autores_afiliacoes');
        const autorCorrespondenteColS = headersSub.indexOf('autor_correspondente');
        const categoriaAutorColS = headersSub.indexOf('categoria_autor');
        const tipoEstudoColS = headersSub.indexOf('tipo_estudo');
        const tipoEstudoOutroColS = headersSub.indexOf('tipo_estudo_outro');
        const aprovacaoEticaColS = headersSub.indexOf('aprovacao_etica');
        const protocoloEticaColS = headersSub.indexOf('protocolo_etica');
        const arquivoNomeColS = headersSub.indexOf('arquivo_nome');
        const arquivoUrlColS = headersSub.indexOf('arquivo_url');
        const arquivoIdentificadoNomeColS = headersSub.indexOf('arquivo_identificado_nome');
        const arquivoIdentificadoUrlColS = headersSub.indexOf('arquivo_identificado_url');
        const arquivoSemIdentificacaoNomeColS = headersSub.indexOf('arquivo_sem_identificacao_nome');
        const arquivoSemIdentificacaoUrlColS = headersSub.indexOf('arquivo_sem_identificacao_url');
        const origemColS = headersSub.indexOf('origem');
        const revisor1ColS = headersSub.indexOf('aceite_revisor1');
        const revisor2ColS = headersSub.indexOf('aceite_revisor2');
        const statusColS = headersSub.indexOf('status');
        for (var l = 1; l < dataSub.length; l++) {
          var cpfSheetS = (cpfColS !== -1 ? dataSub[l][cpfColS] : '').toString().replace(/\D/g, '').padStart(11, '0');
          if (cpfSheetS === cpfBusca) {
            const r1 = revisor1ColS !== -1 ? dataSub[l][revisor1ColS] : '';
            const r2 = revisor2ColS !== -1 ? dataSub[l][revisor2ColS] : '';
            const statusPlanilha = statusColS !== -1 && dataSub[l][statusColS] ? dataSub[l][statusColS] : '';
            let statusSub = statusPlanilha || "Enviado";
            if (r1 === "Sim" && r2 === "Sim") statusSub = "Aprovado";
            else if (r1 === "Não" || r2 === "Não") statusSub = "Não aprovado";
            else if (!statusPlanilha && (r1 || r2)) statusSub = "Em avaliação";

            data.submissoes.push({
              data_envio: dataEnvioColS !== -1 ? dataSub[l][dataEnvioColS] : '',
              titulo: tituloColS !== -1 ? dataSub[l][tituloColS] : '',
              area_tematica: areaTematicaColS !== -1 ? dataSub[l][areaTematicaColS] : '',
              autores_afiliacoes: autoresAfiliacoesColS !== -1 ? dataSub[l][autoresAfiliacoesColS] : '',
              autor_correspondente: autorCorrespondenteColS !== -1 ? dataSub[l][autorCorrespondenteColS] : '',
              categoria_autor: categoriaAutorColS !== -1 ? dataSub[l][categoriaAutorColS] : '',
              tipo_estudo: tipoEstudoColS !== -1 ? dataSub[l][tipoEstudoColS] : '',
              tipo_estudo_outro: tipoEstudoOutroColS !== -1 ? dataSub[l][tipoEstudoOutroColS] : '',
              aprovacao_etica: aprovacaoEticaColS !== -1 ? dataSub[l][aprovacaoEticaColS] : '',
              protocolo_etica: protocoloEticaColS !== -1 ? dataSub[l][protocoloEticaColS] : '',
              arquivo_nome: arquivoNomeColS !== -1 ? dataSub[l][arquivoNomeColS] : '',
              arquivo_url: arquivoUrlColS !== -1 ? dataSub[l][arquivoUrlColS] : '',
              arquivo_identificado_nome: arquivoIdentificadoNomeColS !== -1 ? dataSub[l][arquivoIdentificadoNomeColS] : '',
              arquivo_identificado_url: arquivoIdentificadoUrlColS !== -1 ? dataSub[l][arquivoIdentificadoUrlColS] : '',
              arquivo_sem_identificacao_nome: arquivoSemIdentificacaoNomeColS !== -1 ? dataSub[l][arquivoSemIdentificacaoNomeColS] : '',
              arquivo_sem_identificacao_url: arquivoSemIdentificacaoUrlColS !== -1 ? dataSub[l][arquivoSemIdentificacaoUrlColS] : '',
              origem: origemColS !== -1 ? dataSub[l][origemColS] : '',
              status: statusSub
            });
          }
        }
      }

      return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 7. INSCRIÇÃO NOS CONCURSOS ======
    else if (action === 'inscreverConcurso') {
      const concursosPermitidos = [
        'Minha Tese em 3 Minutos',
        'Concurso de Bolos',
        'Concurso de Imagens de Microscopia'
      ];
      const concurso = (e.parameter.concurso || '').toString().trim();
      const isConcursoImagem = concurso === 'Concurso de Imagens de Microscopia';
      const isConcursoBolo = concurso === 'Concurso de Bolos';
      const nome = (e.parameter.nome || '').toString().trim();
      const cpfLimpo = normalizarCpf_(e.parameter.cpf);
      const email = isEmailValido_(e.parameter.email);
      const telefone = (e.parameter.telefone || '').toString().trim();
      const vinculo = (e.parameter.vinculo || '').toString().trim();
      const origem = (e.parameter.origem || '').toString().trim() || 'Não informado';
      const titulo = (e.parameter.titulo || '').toString().trim();
      const observacoes = (e.parameter.observacoes || '').toString().trim();
      const categoriaImagem = (e.parameter.categoria_imagem || '').toString().trim();
      const legendaImagem = (e.parameter.legenda_imagem || '').toString().trim();
      const termoCessao = (e.parameter.termo_cessao || '').toString().trim();
      const nomeBolo = (e.parameter.nome_bolo || '').toString().trim();
      const ingredientesBolo = (e.parameter.ingredientes_bolo || '').toString().trim();
      const alergenicosBolo = (e.parameter.alergenicos_bolo || '').toString().trim();
      const responsavelBolo = (e.parameter.responsavel_bolo || '').toString().trim();
      const filenameArquivo = (e.parameter.filename || '').toString().trim();
      const fileDataBase64 = (e.parameter.fileData || '').toString().trim();
      const mimeTypeArquivo = e.parameter.mimeType || 'application/octet-stream';

      if (concursosPermitidos.indexOf(concurso) === -1) {
        throw new Error('Concurso inválido.');
      }
      if (!nome || !cpfLimpo || !email || !telefone || !vinculo) {
        throw new Error('Dados obrigatórios da inscrição não enviados.');
      }
      if (!isConcursoBolo && (!filenameArquivo || !fileDataBase64)) {
        throw new Error(isConcursoImagem ? 'Imagem da inscrição não enviada.' : 'Vídeo da inscrição não enviado.');
      }
      if (FOLDER_ID_VIDEOS_CONCURSOS === 'COLE_AQUI_O_ID_DA_PASTA_DE_VIDEOS') {
        throw new Error('Pasta de arquivos dos concursos ainda não configurada no Apps Script.');
      }
      if (isConcursoBolo) {
        const sheetPrincipalBolo = doc.getSheetByName(sheetName);
        const dataPrincipalBolo = sheetPrincipalBolo.getDataRange().getValues();
        const headersPrincipalBolo = dataPrincipalBolo[0].map(function(h) { return h.toString().toLowerCase().trim(); });
        const cpfColBolo = headersPrincipalBolo.indexOf('cpf');
        let participanteVinculado = false;
        if (cpfColBolo !== -1) {
          for (var b = 1; b < dataPrincipalBolo.length; b++) {
            if (normalizarCpf_(dataPrincipalBolo[b][cpfColBolo]) === cpfLimpo) {
              participanteVinculado = true;
              break;
            }
          }
        }
        if (!participanteVinculado) {
          throw new Error('O Concurso de Bolos é exclusivo para participantes vinculados ao evento.');
        }
        if (!titulo) {
          throw new Error('Informe o participante ou grupo responsável pelo bolo.');
        }
        if (!nomeBolo) {
          throw new Error('Informe o nome criativo do bolo.');
        }
        if (!ingredientesBolo) {
          throw new Error('Informe a lista completa de ingredientes do bolo.');
        }
        if (!alergenicosBolo) {
          throw new Error('Informe as restrições alimentares e possíveis alergênicos.');
        }
        if (!responsavelBolo) {
          throw new Error('Informe o responsável pela entrega, identificação e retirada dos materiais.');
        }
      } else if (isConcursoImagem) {
        if (!titulo || contarPalavras_(titulo) > 6) {
          throw new Error('Informe um título da imagem com no máximo 6 palavras.');
        }
        if (!categoriaImagem) {
          throw new Error('Informe a categoria da imagem.');
        }
        if (!legendaImagem || contarPalavras_(legendaImagem) > 100) {
          throw new Error('Informe uma legenda da imagem com no máximo 100 palavras.');
        }
        if (termoCessao !== 'Sim') {
          throw new Error('Confirme o termo de cessão de uso da imagem.');
        }
        if (!/^image\/(jpeg|tiff)$/i.test(mimeTypeArquivo) && !/\.(jpe?g|tiff?)$/i.test(filenameArquivo)) {
          throw new Error('Envie uma imagem em JPEG ou TIFF.');
        }
      } else if (!/^video\//i.test(mimeTypeArquivo)) {
        throw new Error('Envie um arquivo de vídeo válido.');
      }

      let arquivoFile = null;
      let arquivoUrl = '';
      if (!isConcursoBolo) {
        const folderVideosConcursos = DriveApp.getFolderById(FOLDER_ID_VIDEOS_CONCURSOS);
        const fileData = Utilities.base64Decode(fileDataBase64);
        if (!fileData || fileData.length === 0) {
          throw new Error(isConcursoImagem ? 'O arquivo de imagem está vazio.' : 'O arquivo de vídeo está vazio.');
        }
        const arquivoBlob = Utilities.newBlob(fileData, mimeTypeArquivo, 'CONCURSO - ' + concurso + ' - ' + nome + ' - ' + filenameArquivo);
        arquivoFile = folderVideosConcursos.createFile(arquivoBlob);
        arquivoUrl = arquivoFile.getUrl();
      }

      const concursosHeaders = ['Data', 'Concurso', 'Nome', 'CPF', 'Vinculo', 'Origem', 'Email', 'Telefone', 'Titulo', 'Observacoes', 'Video_nome', 'Video_url', 'Status', 'Categoria_imagem', 'Legenda_imagem', 'Termo_cessao', 'Nome_bolo', 'Ingredientes_bolo', 'Alergenicos_bolo', 'Responsavel_bolo'];
      let sheetConcursos = doc.getSheetByName('concursos');
      if (!sheetConcursos) {
        sheetConcursos = doc.insertSheet('concursos');
      }
      if (sheetConcursos.getMaxColumns() < concursosHeaders.length) {
        sheetConcursos.insertColumnsAfter(sheetConcursos.getMaxColumns(), concursosHeaders.length - sheetConcursos.getMaxColumns());
      }
      if (sheetConcursos.getLastRow() === 0) {
        sheetConcursos.appendRow(concursosHeaders);
      } else {
        sheetConcursos.getRange(1, 1, 1, concursosHeaders.length).setValues([concursosHeaders]);
      }

      const dataConcursos = sheetConcursos.getDataRange().getValues();
      let rowToUpdate = -1;
      for (var i = dataConcursos.length - 1; i > 0; i--) {
        const cpfSheet = normalizarCpf_(dataConcursos[i][3]);
        const concursoSheet = (dataConcursos[i][1] || '').toString().trim();
        const categoriaSheet = (dataConcursos[i][13] || '').toString().trim();
        if (cpfSheet === cpfLimpo && concursoSheet === concurso && (!isConcursoImagem || categoriaSheet === categoriaImagem)) {
          rowToUpdate = i + 1;
          break;
        }
      }

      const rowData = [
        new Date(),
        concurso,
        nome,
        "'" + cpfLimpo,
        vinculo,
        origem,
        email,
        telefone,
        titulo,
        observacoes,
        arquivoFile ? arquivoFile.getName() : '',
        arquivoUrl,
        'Inscrito',
        categoriaImagem,
        legendaImagem,
        termoCessao,
        nomeBolo,
        ingredientesBolo,
        alergenicosBolo,
        responsavelBolo
      ];

      if (rowToUpdate !== -1) {
        sheetConcursos.getRange(rowToUpdate, 1, 1, concursosHeaders.length).setValues([rowData]);
      } else {
        sheetConcursos.appendRow(rowData);
      }

      try {
        enviarEmailConfirmacaoConcurso_(email, nome, concurso);
      } catch (emailError) {
        console.error('Erro ao enviar confirmação de inscrição no concurso para ' + email + ': ' + emailError.toString());
      }

      return ContentService.createTextOutput(JSON.stringify({ result: 'success', updated: rowToUpdate !== -1 })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 7.1. BUSCAR DADOS DO INSCRITO PARA CONCURSOS ======
    else if (action === 'buscarInscritoConcurso') {
      const cpfBusca = normalizarCpf_(e.parameter.cpf);
      const sheetVerif = doc.getSheetByName(sheetName);
      const data = sheetVerif.getDataRange().getValues();
      const headers = data[0].map(function(h) { return h.toString().toLowerCase().trim(); });
      const cpfCol = headers.indexOf('cpf');
      const nomeCol = headers.indexOf('nome');
      const emailCol = headers.indexOf('email');
      const telefoneCol = headers.indexOf('telefone');

      if (!cpfBusca || cpfCol === -1) {
        return ContentService.createTextOutput(JSON.stringify({ encontrado: false })).setMimeType(ContentService.MimeType.JSON);
      }

      for (var i = data.length - 1; i > 0; i--) {
        const cpfSheet = normalizarCpf_(data[i][cpfCol]);
        if (cpfSheet === cpfBusca) {
          return ContentService.createTextOutput(JSON.stringify({
            encontrado: true,
            cpf: cpfBusca,
            nome: nomeCol !== -1 ? data[i][nomeCol] || '' : '',
            email: emailCol !== -1 ? data[i][emailCol] || '' : '',
            telefone: telefoneCol !== -1 ? data[i][telefoneCol] || '' : ''
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }

      return ContentService.createTextOutput(JSON.stringify({ encontrado: false })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 8. UPLOAD DE COMPROVANTE (Produtos) ======
    else if (action === 'uploadFesta') {
      const folder = DriveApp.getFolderById(FOLDER_ID);
      if (!e.parameter.fileData || !e.parameter.filename) {
        throw new Error('Comprovante não enviado.');
      }
      const fileData = Utilities.base64Decode(e.parameter.fileData);
      if (!fileData || fileData.length === 0) {
        throw new Error('O arquivo do comprovante está vazio.');
      }
      const tipoProduto = (e.parameter.tipo || '').toString().trim();
      if (!tipoProduto) {
        throw new Error('Tipo do produto não informado.');
      }
      const quantidadeProduto = Math.max(1, parseInt(e.parameter.quantidade || '1', 10) || 1);
      const tipoQuantidade = tipoProduto + ' - quantidade: ' + quantidadeProduto;
      const blob = Utilities.newBlob(fileData, e.parameter.mimeType, "PRODUTO - " + tipoProduto + " - " + e.parameter.nome + " - " + e.parameter.filename);
      const file = folder.createFile(blob);
      let sheetProdutos = doc.getSheetByName('produtos');
      const produtosHeaders = ['Data', 'Nome', 'CPF', 'Tipo_quantidade', 'Valor', 'Comprovante', 'email', 'telefone', 'Observacoes'];
      if (!sheetProdutos) {
        sheetProdutos = doc.insertSheet('produtos');
      }
      if (sheetProdutos.getMaxColumns() < produtosHeaders.length) {
        sheetProdutos.insertColumnsAfter(sheetProdutos.getMaxColumns(), produtosHeaders.length - sheetProdutos.getMaxColumns());
      }
      if (sheetProdutos.getMaxColumns() > produtosHeaders.length) {
        sheetProdutos.deleteColumns(produtosHeaders.length + 1, sheetProdutos.getMaxColumns() - produtosHeaders.length);
      }
      if (sheetProdutos.getLastRow() === 0) {
        sheetProdutos.appendRow(produtosHeaders);
      } else {
        sheetProdutos.getRange(1, 1, 1, produtosHeaders.length).setValues([produtosHeaders]);
      }
      var cpfParamFesta = (e.parameter.cpf || '').toString().replace(/\D/g, '');
      var cpfFesta = cpfParamFesta ? "'" + cpfParamFesta.padStart(11, '0') : '';

      sheetProdutos.appendRow([
        new Date(), 
        e.parameter.nome, 
        cpfFesta, 
        tipoQuantidade, 
        'R$ ' + e.parameter.valor, 
        file.getUrl(),
        e.parameter.email || '',
        e.parameter.telefone || '',
        e.parameter.observacoes || ''
      ]);

      return ContentService.createTextOutput(JSON.stringify({ 'result': 'success', 'fileUrl': file.getUrl() })).setMimeType(ContentService.MimeType.JSON);
    }

    // ====== 9. INSCRIÇÃO NORMAL (Dados do Formulário) ======
    else {
      return ContentService.createTextOutput(JSON.stringify({
        result: 'error',
        message: 'A inscrição só é gravada após o envio do comprovante.'
      })).setMimeType(ContentService.MimeType.JSON);
    }

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ 'result': 'error', 'error': error.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    if (lockAcquired) lock.releaseLock();
  }
}
